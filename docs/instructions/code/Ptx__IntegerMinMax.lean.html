9theorem min_toNat (a b : Word) : (BinOp.eval .minU a b).toNat = Nat.min a.toNat b.toNat := by
10  by_cases h : a.toNat ≤ b.toNat <;> simp [BinOp.eval, h, Nat.min_def]

12theorem max_toNat (a b : Word) : (BinOp.eval .maxU a b).toNat = Nat.max a.toNat b.toNat := by

41theorem min_bounds (a b : Word) :

51/-- `eval`'s bin32 branch computes both operands from the incoming state, then
52updates only the destination word register. The record equality also exposes
53the unchanged PC successor, address registers, predicates, and memory. -/
54theorem min_exec (guard : Guard) (s : State) (d : Nat) (a b : Operand32)
55    (hg : guard.eval s = true) :
56    eval none ⟨guard, .bin32 .minU d a b⟩ s =
57      .next ⟨s.pc + 1, update s.regs d (BinOp.eval .minU (a.eval s) (b.eval s)),
58        s.addrs, s.preds, s.memory⟩
59        (occurrence s ⟨guard, .bin32 .minU d a b⟩ true) := by
60  simp [eval, hg]

70/-- The actual evaluated state leaves every non-destination word register at
71its incoming value. `update_other` is the shared register-update fact used. -/
72theorem min_exec_preserves_other (guard : Guard) (s : State) (d i : Nat)
73    (a b : Operand32) (hg : guard.eval s = true) (hi : i ≠ d) :
74    (match eval none ⟨guard, .bin32 .minU d a b⟩ s with
75      | .next t _ => t.regs i | _ => s.regs i) = s.regs i := by
76  rw [min_exec guard s d a b hg]
77  exact update_other hi
