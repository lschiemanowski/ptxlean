11def producer : List Instr := [.store .relaxed 0 7, .store .release 1 1]
12def consumer (order : LoadOrder) : List Instr := [.load order 1 0, .load .relaxed 0 1]
13def program (order : LoadOrder) : Program := ⟨[0, 0], [producer, consumer order]⟩
14def registers : Nat → Registers := fun _ _ => 0

16def oracle (flag payload : Word) (tid position : Nat) : Word :=
17  if tid = 1 then if position = 0 then flag else payload else 0

19/-- Indices: initial payload, initial flag, payload store, release store,
20flag load, payload load. This enumeration is generated, not an input trace. -/
21def graph (order : LoadOrder) (flag payload : Word)
22    (source : Fin 6 → Fin 6) (co : Fin 6 → Fin 6 → Bool) : Graph 6 :=
23  (program order).graph registers (oracle flag payload) source co

50/-- The publication property quantifies over all compatible source and coherence
51choices, rather than selecting one witness or assuming the desired payload equality. -/
52theorem publication (payload : Word) (source co)
53    (valid : (graph .acquire 1 payload source co).Valid) : payload = 7 := by
54  let g := graph .acquire 1 payload source co
55  have flagSource : source c = b := by
56    have h := valid.sources.compatible c (by trivial)
57    change g.write (source c) ∧ g.sameAddress (source c) c ∧
58      (g.event (source c)).effect.value = 1 at h
59    have cases : source c = 0 ∨ source c = 1 ∨ source c = 2 ∨
60        source c = 3 ∨ source c = 4 ∨ source c = 5 := by omega
61    rcases cases with h' | h' | h' | h' | h' | h'
62    · rw [h'] at h
63      have bad : (0 : Word) = 1 := h.2.2
64      exact False.elim (by contradiction)
65    · rw [h'] at h
66      have bad : (0 : Word) = 1 := h.2.2
67      exact False.elim (by contradiction)
68    · rw [h'] at h
69      have bad : (0 : Nat) = 1 := h.2.1
70      omega
71    · exact h'
72    · rw [h'] at h
73      exact False.elim h.1
74    · rw [h'] at h
75      exact False.elim h.1
76  have synchronizes : g.sync b c := by
77    refine ⟨by change some 0 ≠ some 1; decide, b, c, Or.inl ⟨rfl, rfl⟩, Or.inl ⟨rfl, rfl⟩, ?_, ?_⟩
78    · exact ⟨⟨by trivial, flagSource⟩, by exact ⟨(by intro h; cases h), (by intro h; cases h), rfl⟩⟩
79    · exact ⟨(by intro h; cases h), (by intro h; cases h), rfl⟩
80  have ordered : g.base a d :=
81    .join (.edge (Or.inl (by exact ⟨(by intro h; cases h), rfl, Nat.zero_lt_one⟩)))
82      (.join (.edge (Or.inr synchronizes)) (.edge (Or.inl (by exact ⟨(by intro h; cases h), rfl, Nat.zero_lt_one⟩))))
83  have causes : g.cause a d := Or.inl ⟨ordered, rfl⟩
84  have notInitial : source d ≠ ip := by
85    intro hs
86    have hc := valid.co.initFirst ip a rfl (by trivial) rfl (by decide)
87    exact valid.no_stale a d (by trivial) (by trivial) rfl causes (by change g.coherence (source d) a; rw [hs]; exact hc)
88  have h := valid.sources.compatible d (by trivial)
89  change g.write (source d) ∧ g.sameAddress (source d) d ∧
90    (g.event (source d)).effect.value = payload at h
91  have cases : source d = 0 ∨ source d = 1 ∨ source d = 2 ∨
92      source d = 3 ∨ source d = 4 ∨ source d = 5 := by omega
93  rcases cases with h' | h' | h' | h' | h' | h'
94  · exact False.elim (notInitial h')
95  · rw [h'] at h
96    have bad : (1 : Nat) = 0 := h.2.1
97    omega
98  · rw [h'] at h
99    exact h.2.2.symm
100  · rw [h'] at h
101    have bad : (1 : Nat) = 0 := h.2.1
102    omega
103  · rw [h'] at h
104    exact False.elim h.1
105  · rw [h'] at h
106    exact False.elim h.1

108/-- Only nontrivial coherence edges are initialization before its program store. -/
109def witnessCo (x y : Fin 6) : Bool := (x == ip && y == a) || (x == iff && y == b)
110def successSource (x : Fin 6) : Fin 6 := if x = c then b else if x = d then a else ip
111def staleSource (x : Fin 6) : Fin 6 := if x = c then b else ip
112
113def success : Graph 6 := graph .acquire 1 7 successSource witnessCo
114def stale : Graph 6 := graph .relaxed 1 0 staleSource witnessCo

116def successUpper (x y : Fin 6) : Prop := 2 ≤ x.val ∧ x.val < y.val
117def staleUpper (x y : Fin 6) : Prop := (x = a ∧ y = b) ∨ (x = c ∧ y = d)
118def staleRank (x : Fin 6) : Nat := if x = a ∨ x = c then 2 else if x = b ∨ x = d then 1 else 0

158/-- Observable values are the actual final consumer registers. -/
159def result (order : LoadOrder) (flag payload : Word) : Word × Word :=
160  let final := (execute (consumer order) (registers 1) (oracle flag payload 1)).1
161  (final 0, final 1)

163theorem result_eq (order : LoadOrder) (flag payload : Word) :
164    result order flag payload = (flag, payload) := rfl

174/-- Two completed local runs, coupled to the same oracle used by graph labels. -/
175theorem local_completion (order : LoadOrder) (flag payload : Word) :
176    Runs producer (registers 0) (oracle flag payload 0)
177      (execute producer (registers 0) (oracle flag payload 0)).1
178      (execute producer (registers 0) (oracle flag payload 0)).2 ∧
179    Runs (consumer order) (registers 1) (oracle flag payload 1)
180      (execute (consumer order) (registers 1) (oracle flag payload 1)).1
181      (execute (consumer order) (registers 1) (oracle flag payload 1)).2 :=
182  ⟨execute_runs _ _ _, execute_runs _ _ _⟩

184/-- Existence includes the actual completed program and final register result. -/
185theorem successful_execution_exists : ∃ (flag payload : Word), ∃ source co,
186    (program .acquire).Admitted registers (oracle flag payload) source co ∧
