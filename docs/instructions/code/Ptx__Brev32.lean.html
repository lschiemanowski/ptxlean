32def lower (i : Instr) : Pure32.Instr family where

34  operation := i.operation

71theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
72    (enabled : i.guard.eval s = true) :
73    Eval i s next event ↔
74      next = Pure32.write s i.destination (result i s) ∧ event = occurrence s i true := by
75  change Pure32.Eval (lower i) s next event ↔ _
76  rw [Pure32.eval_true_iff (lower i) s next event enabled]
77  constructor
78  · rintro ⟨value, hres, hnext, hevent⟩
79    have hv : value = result i s := by
80      change value = compute i.operation ((lower i).wordValues s w0) at hres
81      simpa [lower_wordValues, result] using hres
82    subst value
83    exact ⟨by simpa [lower, result, Pure32.Instr.wordValues] using hnext,
84      by simpa [occurrence, lower] using hevent⟩
85  · rintro ⟨hnext, hevent⟩
86    refine ⟨result i s, ?_, hnext, hevent⟩
87    change result i s = compute i.operation ((lower i).wordValues s w0)
88    simp [lower_wordValues, result]

126theorem step_origin (target : Ptx.Target) (program : List Instr)
127    (s next : Scalar.State) (event : Occurrence) (stepped : Step target program s next event) :
128    SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧
129      event.pc = s.pc ∧ event.instruction = lower i ∧ Eval i s next event := by
130  obtain ⟨j, fetch, supported, pc, ins, evaluated⟩ := Pure32.step_origin stepped
131  have original : ∃ i, program[s.pc]? = some i ∧ lower i = j := by
132    cases hx : program[s.pc]? with
133    | none => simp [hx] at fetch
134    | some i =>
135      have hm : some (lower i) = some j := by simpa [hx] using fetch
136      exact ⟨i, by simp, Option.some.inj hm⟩
137  obtain ⟨i, fetchi, eq⟩ := original
138  have ev : Eval i s next event :=
139    Eq.mp (congrArg (fun x : Pure32.Instr family => Pure32.Eval x s next event) eq.symm) evaluated
140  exact ⟨supported, i, fetchi, pc, ins.trans eq.symm, ev⟩

142theorem step_exists (target : Ptx.Target) (program : List Instr) (s : Scalar.State) (i : Instr)
143    (supported : SupportedTarget target) (fetch : program[s.pc]? = some i) :
144    ∃ next event, Step target program s next event := by
145  have mapped : (program.map lower)[s.pc]? = some (lower i) := by simp [fetch]
146  apply Pure32.step_exists (target := target) (program := program.map lower) (i := lower i)
147    (s := s) (fetch := mapped)
148  exact supported

171def encode (i : Instr) : Scalar.Text.Statement :=

174def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
175  if statement.mnemonic = "brev.b32" then
176    match statement.operands with
177    | [.word (.reg d), .word a] => .ok ⟨statement.guard, .brev, d, a⟩
178    | _ => .error (.invalidOperands statement.mnemonic)
179  else .error (.unsupportedMnemonic statement.mnemonic)

181theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
182  cases i with
183  | mk guard operation destination source => cases operation <;> rfl

185theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) :
186    decode statement = .ok i ↔ statement = encode i := by
187  constructor
188  · intro h
189    rcases i with ⟨g, op, d, a⟩
190    cases op <;> cases statement with
191      | mk guard spelling operands =>
192        unfold decode at h
193        split at h
194        all_goals first
195          | (split at h <;> simp_all [encode, mnemonic])
196          | (cases h)
197  · intro h
198    subst statement
199    exact decode_encode i
