3/-! Reusable concrete scalar verification rules. Segment exhaustion, successful
4completion, and partial correctness are distinct contracts. Frame and invariant
5rules also apply to candidate reads; they do not admit those candidates under a
6concurrent PTX memory model. -/
7namespace Ptx.Scalar.Rules

9abbrev Assertion := State → Prop
10
11/-- An exact finite prefix reaches an intermediate assertion without terminating. -/
12def Segment (program : List Instr) (fuel : Nat) (pre post : Assertion) : Prop :=
13  ∀ s, pre s → (run fuel program s).status = .exhausted ∧ post (run fuel program s).state
14
15/-- A fixed-budget total-correctness contract: explicit exit and the postcondition. -/
16def Completed (program : List Instr) (fuel : Nat) (pre post : Assertion) : Prop :=
17  ∀ s, pre s → (run fuel program s).status = .halted ∧ post (run fuel program s).state
18
19/-- No termination claim: this constrains only runs that explicitly halt. -/
20def PartialCorrect (program : List Instr) (pre post : Assertion) : Prop :=
21  ∀ s, pre s → ∀ fuel, (run fuel program s).status = .halted → post (run fuel program s).state
22
23/-- Termination with a state-dependent finite budget. -/
24def TotalCorrect (program : List Instr) (pre post : Assertion) : Prop :=
25  ∀ s, pre s → ∃ fuel, (run fuel program s).status = .halted ∧ post (run fuel program s).state

27theorem segment_comp (first : Segment program firstFuel pre middle)
28    (second : Segment program secondFuel middle post) :
29    Segment program (firstFuel + secondFuel) pre post := by
30  intro s hs
31  obtain ⟨hf, hm⟩ := first s hs
32  obtain ⟨hh, hp⟩ := second _ hm
33  rw [run_add]
34  simpa only [resume, hf, ↓reduceIte] using And.intro hh hp

45theorem segment_then_total (first : Segment program firstFuel pre middle)
46    (second : TotalCorrect program middle post) : TotalCorrect program pre post := by
47  intro s hs
48  obtain ⟨hf, hm⟩ := first s hs
49  obtain ⟨secondFuel, hh, hp⟩ := second _ hm
50  refine ⟨firstFuel + secondFuel, ?_⟩
51  rw [run_add]
52  simpa only [resume, hf, ↓reduceIte] using And.intro hh hp

54theorem completed_total (h : Completed program fuel pre post) :
55    TotalCorrect program pre post := fun s hs => ⟨fuel, h s hs⟩

57/-- A trace records a write to a word only when an actual executed store emits it. -/
58def StoresTo (event : Occurrence) (index : Nat) : Prop :=
59  ∃ effect, event.memory = some effect ∧ effect.kind = .store ∧ effect.address.toNat / 4 = index
60
61def TraceAvoids (trace : List Occurrence) (index : Nat) : Prop :=
62  ∀ event ∈ trace, ¬StoresTo event index
63
64theorem eval_frame (hstep : eval override instruction s = .next next event)
65    (untouched : ¬StoresTo event index) : next.memory[index]? = s.memory[index]? := by
66  cases instruction with
67  | mk guard op =>
68    cases op <;> cases hguard : guard.eval s <;>
69      simp only [eval, hguard, ↓reduceIte] at hstep
70    all_goals first
71      | (obtain ⟨rfl, rfl⟩ := StepResult.next.inj hstep; rfl)
72      | contradiction
73      | skip
74    all_goals
75      split at hstep
76      · contradiction
77      · obtain ⟨rfl, rfl⟩ := StepResult.next.inj hstep
78        first
79        | rfl
80        | apply List.getElem?_set_ne
81          intro equal
82          apply untouched
83          refine ⟨_,rfl,rfl,?_⟩
84          have address := addressIndex_ok_iff.mp (by assumption)
85          exact address.2.symm.trans equal

87theorem step_frame (hstep : stepWith override program s = .next next event)
88    (untouched : ¬StoresTo event index) : next.memory[index]? = s.memory[index]? := by
89  unfold stepWith at hstep
90  split at hstep
91  · contradiction
92  · exact eval_frame hstep untouched

94/-- Outside actual emitted stores, all words are preserved, including on fault,
95unsupported, exhausted, and candidate-read executions. No final-memory premise. -/
96theorem runWith_frame (fuel : Nat) (oracle : Nat → Option Word) (program : List Instr)
97    (s : State) (index : Nat)
98    (untouched : TraceAvoids (runWith fuel oracle program s).trace index) :
99    (runWith fuel oracle program s).state.memory[index]? = s.memory[index]? := by
100  induction fuel generalizing oracle s with
101  | zero => rfl
102  | succ fuel ih =>
103    cases hs : stepWith (oracle 0) program s with
104    | next next emitted =>
105      simp only [runWith, hs] at untouched ⊢
106      have head := untouched emitted (by simp)
107      have tail : TraceAvoids (runWith fuel (fun i => oracle (i + 1)) program next).trace index :=
108        fun event member => untouched event (by simp [member])
109      exact (ih _ _ tail).trans (step_frame hs head)
110    | halted final emitted =>
111      simp only [runWith, hs]
112      rw [(step_halted hs).1]
113    | fault reason => simp [runWith, hs]
114    | unsupported spelling => simp [runWith, hs]

116theorem run_frame (fuel : Nat) (program : List Instr) (s : State) (index : Nat)
117    (untouched : TraceAvoids (run fuel program s).trace index) :
118    (run fuel program s).state.memory[index]? = s.memory[index]? :=
119  runWith_frame fuel (fun _ => none) program s index untouched

121/-- A one-step invariant lifts to every finite prefix. Terminal faults leave the
122state unchanged, and explicit exit also leaves its incoming state unchanged. -/
123theorem runWith_invariant (invariant : Assertion)
124    (preserved : ∀ override s next event, invariant s →
125      stepWith override program s = .next next event → invariant next)
126    (fuel : Nat) (oracle : Nat → Option Word) (s : State) (initial : invariant s) :
127    invariant (runWith fuel oracle program s).state := by
128  induction fuel generalizing oracle s with
129  | zero => exact initial
130  | succ fuel ih =>
131    cases hs : stepWith (oracle 0) program s with
132    | next next emitted =>
133      simp only [runWith, hs]
134      exact ih _ _ (preserved _ _ _ _ initial hs)
135    | halted final emitted =>
136      simp only [runWith, hs]
137      rw [(step_halted hs).1]
138      exact initial
139    | fault reason => simpa [runWith, hs] using initial
140    | unsupported spelling => simpa [runWith, hs] using initial

142/-- Invariant preservation and the exit case establish partial correctness.
143No progress or decreasing-measure premise is required, so divergence remains possible. -/
144theorem partial_of_invariant (program : List Instr) (pre invariant post : Assertion)
145    (initial : ∀ s, pre s → invariant s)
146    (preserved : ∀ s next event, invariant s →
147      step program s = .next next event → invariant next)
148    (exitPost : ∀ s final event, invariant s →
149      step program s = .halted final event → post final) :
150    PartialCorrect program pre post := by
151  have aux : ∀ fuel s, invariant s → (run fuel program s).status = .halted →
152      post (run fuel program s).state := by
153    intro fuel
154    induction fuel with
155    | zero => intro s hs halted; simp [run_zero] at halted
156    | succ fuel ih =>
157      intro s hs halted
158      rw [run_succ] at halted ⊢
159      cases stepEq : step program s with
160      | next next event =>
161        simp only [stepEq] at halted ⊢
162        exact ih next (preserved s next event hs stepEq) halted
163      | halted final event =>
164        exact exitPost s final event hs stepEq
165      | fault reason => simp [stepEq] at halted
166      | unsupported spelling => simp [stepEq] at halted
167  exact fun s hs fuel halted => aux fuel s (initial s hs) halted

169/-- A decreasing natural measure and a local progress obligation establish
170termination. The progress premise excludes faults/unsupported outcomes without
171assuming any completed run; it may be discharged instruction by instruction. -/
172theorem terminates_of_decreasing_measure (program : List Instr) (invariant post : Assertion)
173    (measure : State → Nat)
174    (progress : ∀ s, invariant s →
175      (∃ final event, step program s = .halted final event ∧ post final) ∨
176      (∃ next event, step program s = .next next event ∧ invariant next ∧
177        measure next < measure s)) :
178    TotalCorrect program invariant post := by
179  have terminate : ∀ rank, ∀ s, measure s = rank → invariant s →
180      ∃ fuel, (run fuel program s).status = .halted ∧ post (run fuel program s).state := by
181    intro rank
182    induction rank using Nat.strongRecOn with
183    | ind rank ih =>
184      intro s rankEq initial
185      rcases progress s initial with ⟨final,event,hs,hpost⟩ | ⟨next,event,hs,hnext,less⟩
186      · exact ⟨1, by simp [run_succ, hs, hpost]⟩
187      · obtain ⟨fuel,hh,hp⟩ := ih (measure next) (by omega) next rfl hnext
188        exact ⟨fuel + 1, by simpa only [run_succ, hs] using And.intro hh hp⟩
189  exact fun s initial => terminate (measure s) s rfl initial
