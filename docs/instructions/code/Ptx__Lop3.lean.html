15structure Instr where

24def compute (op : Operation) (a b c : Word) : Word :=
25  let na := ~~~a
26  let nb := ~~~b
27  let nc := ~~~c
28  (if op.getLsbD 0 then na &&& nb &&& nc else 0) |||
29  (if op.getLsbD 1 then na &&& nb &&& c else 0) |||
30  (if op.getLsbD 2 then na &&& b &&& nc else 0) |||
31  (if op.getLsbD 3 then na &&& b &&& c else 0) |||
32  (if op.getLsbD 4 then a &&& nb &&& nc else 0) |||
33  (if op.getLsbD 5 then a &&& nb &&& c else 0) |||
34  (if op.getLsbD 6 then a &&& b &&& nc else 0) |||
35  (if op.getLsbD 7 then a &&& b &&& c else 0)

37def family : Pure32.Family Operation :=
38  Pure32.Family.ofFunction (fun _ => 3) (fun _ => 0)
39    (fun op words _ => compute op (words w0) (words w1) (words w2))
40    (fun target _ => SupportedTarget target)

42def lower (i : Instr) : Pure32.Instr family where
43  guard := i.guard
44  operation := i.operation
45  destination := i.destination
46  words := fun j => if j.val = 0 then i.first else if j.val = 1 then i.second else i.third
47  predicates := Fin.elim0

49def result (i : Instr) (s : Scalar.State) : Word :=
50  compute i.operation (i.first.eval s) (i.second.eval s) (i.third.eval s)

59private theorem observe_if (condition : Bool) (value : Word) (bit : Nat) :
60    (if condition then value else (0 : Word)).getLsbD bit = (condition && value.getLsbD bit) :=

63theorem compute_bit (op : Operation) (a b c : Word) (bit : Nat) (bound : bit < 32) :
64    (compute op a b c).getLsbD bit = (op.getLsbD ((if a.getLsbD bit then 4 else 0) + (if b.getLsbD bit then 2 else 0) + (if c.getLsbD bit then 1 else 0))) := by
65  simp only [compute, BitVec.getLsbD_or, observe_if, BitVec.getLsbD_and,
66    BitVec.getLsbD_not, BitVec.getLsbD_ofNat, bound, Bool.true_and, Bool.false_and]
67  cases ha : a.getLsbD bit <;> cases hb : b.getLsbD bit <;> cases hc : c.getLsbD bit <;>
68    simp [ha, hb, hc]

86theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
87    (enabled : i.guard.eval s = true) :
88    Eval i s next event ↔
89      next = Pure32.write s i.destination (result i s) ∧ event = occurrence s i true := by
90  change Pure32.Eval (lower i) s next event ↔ _
91  rw [Pure32.eval_true_iff (lower i) s next event enabled]
92  constructor
93  · rintro ⟨value, hres, hnext, hevent⟩
94    have hv : value = result i s := by
95      simpa [family, lower, Pure32.Family.ofFunction, Pure32.Instr.wordValues,
96        Pure32.Instr.predicateValues, result, w0, w1, w2] using hres
97    subst value
98    exact ⟨hnext, hevent⟩
99  · rintro ⟨hnext, hevent⟩
100    refine ⟨result i s, ?_, hnext, hevent⟩
101    simp [family, lower, Pure32.Family.ofFunction, Pure32.Instr.wordValues,
102      result, w0, w1, w2]

140theorem step_origin (target : Ptx.Target) (program : List Instr)
141    (s next : Scalar.State) (event : Occurrence) (stepped : Step target program s next event) :
142    SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧
143      event.pc = s.pc ∧ event.instruction = lower i ∧ Eval i s next event := by
144  obtain ⟨j, fetch, supported, pc, ins, evaluated⟩ := Pure32.step_origin stepped
145  have original : ∃ i, program[s.pc]? = some i ∧ lower i = j := by
146    cases hx : program[s.pc]? with
147    | none => simp [hx] at fetch
148    | some i =>
149      have hm : some (lower i) = some j := by simpa [hx] using fetch
150      exact ⟨i, by simp, Option.some.inj hm⟩
151  obtain ⟨i, fetchi, eq⟩ := original
152  have ev : Eval i s next event :=
153    Eq.mp (congrArg (fun x : Pure32.Instr family => Pure32.Eval x s next event) eq.symm) evaluated
154  exact ⟨supported, i, fetchi, pc, ins.trans eq.symm, ev⟩

178def encode (i : Instr) : Scalar.Text.Statement :=
179  ⟨i.guard, "lop3.b32", [.word (.reg i.destination), .word i.first, .word i.second,
180    .word i.third, .word (.imm (BitVec.ofNat 32 i.operation.toNat))]⟩

182def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
183  match statement.mnemonic, statement.operands with
184  | "lop3.b32", [.word (.reg d), .word a, .word b, .word c, .word (.imm lut)] =>
185      if lut.toNat ≤ 255 then .ok ⟨statement.guard, BitVec.ofNat 8 lut.toNat, d, a, b, c⟩
186      else .error (.invalidOperands "lop3.b32")
187  | spelling, _ => if supportedMnemonic spelling then .error (.invalidOperands spelling)
188      else .error (.unsupportedMnemonic spelling)

190theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
191  cases i with
192  | mk guard operation destination first second third =>
193    have hnat : (BitVec.ofNat 32 operation.toNat).toNat = operation.toNat :=
194      Ptx.BitVecProof.toNat_widen operation (by decide)
195    have hbound : operation.toNat ≤ 255 := by
196      have hlt := operation.isLt
197      omega
198    simp only [encode, decode, hnat, if_pos hbound]
199    have hop : BitVec.ofNat 8 operation.toNat = operation := by
200      rw [BitVec.ofNat_toNat]
201      simp
202    simp [hop]

204theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) :
205    decode statement = .ok i ↔ statement = encode i := by
206  constructor
207  · intro h
208    cases i with
209    | mk g op d a b c =>
210      cases statement with
211      | mk guard spelling operands =>
212        simp only [decode] at h
213        split at h
214        · rename_i lut
215          split at h
216          · injection h with hinj
217            rcases hinj with ⟨rfl, hop, rfl, rfl, rfl, rfl⟩
218            have hwide : BitVec.ofNat 32 (BitVec.ofNat 8 lut.toNat).toNat = lut :=
219              Ptx.BitVecProof.widen_narrow lut (by omega)
220            have hEq : lut = BitVec.ofNat 32 (lut.toNat % 256) := by
221              apply BitVec.eq_of_toNat_eq
222              simp only [BitVec.toNat_ofNat]
223              omega
224            simp only [encode]
225            rw [hEq]
226            have hbnd : lut.toNat < 256 := by omega
227            simp [Nat.mod_eq_of_lt hbnd]
228          · simp at h
229        · split at h <;> cases h
230  · intro h
231    subst statement
232    exact decode_encode i
