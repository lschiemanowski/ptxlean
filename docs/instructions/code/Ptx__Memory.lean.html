10/-- Nonempty finite relational path. No bounded search or scheduler is assumed. -/
11inductive Path (r : α → α → Prop) : α → α → Prop where
12  | edge : r a b → Path r a b
13  | join : Path r a b → Path r b c → Path r a c
14
15namespace Path
16
17variable {α : Type u} {r s : α → α → Prop} {a b : α}
18
19theorem contained (edge : ∀ a b, r a b → s a b)
20    (trans : ∀ a b c, s a b → s b c → s a c)
21    (h : Path r a b) : s a b := by
22  induction h with
23  | edge h => exact edge _ _ h
24  | join _ _ ih₁ ih₂ => exact trans _ _ _ ih₁ ih₂
25
26theorem rank_increases (rank : α → Nat)
27    (increases : ∀ a b, r a b → rank a < rank b)
28    (h : Path r a b) : rank a < rank b := by
29  exact contained (s := fun x y => rank x < rank y) increases
30    (fun _ _ _ hab hbc => Nat.lt_trans hab hbc) h
31
32end Path

34structure Occurrence where
35  thread : Option Nat
36  position : Nat
37  effect : Effect
38  deriving DecidableEq, Repr

40/-- Fixed event labels; the completed local traces supply these labels. -/
41structure Graph (n : Nat) where
42  event : Fin n → Occurrence
43  source : Fin n → Fin n
44  co : Fin n → Fin n → Bool

46namespace Graph

48def read (g : Graph n) (e : Fin n) : Prop :=
49  match (g.event e).effect.op with | .load _ => True | _ => False
50
51def write (g : Graph n) (e : Fin n) : Prop :=
52  match (g.event e).effect.op with | .store _ | .init => True | _ => False
53
54instance (g : Graph n) (e : Fin n) : Decidable (g.read e) := by
55  unfold read
56  split <;> infer_instance
57
58instance (g : Graph n) (e : Fin n) : Decidable (g.write e) := by
59  unfold write
60  split <;> infer_instance
61
62def initial (g : Graph n) (e : Fin n) : Prop := (g.event e).effect.op = .init
63
64def acquire (g : Graph n) (e : Fin n) : Prop :=
65  (g.event e).effect.op = .load .acquire
66
67def release (g : Graph n) (e : Fin n) : Prop :=
68  (g.event e).effect.op = .store .release
69
70def sameAddress (g : Graph n) (a b : Fin n) : Prop :=
71  (g.event a).effect.address = (g.event b).effect.address
72
73/-- §8.9.1: only events of a single program thread, ordered by occurrence. -/
74def po (g : Graph n) (a b : Fin n) : Prop :=
75  (g.event a).thread ≠ none ∧ (g.event a).thread = (g.event b).thread ∧
76  (g.event a).position < (g.event b).position
77
78/-- §8.7 restricted to strong GPU/generic program accesses with full overlap. -/
79def morallyStrong (g : Graph n) (a b : Fin n) : Prop :=
80  ¬g.initial a ∧ ¬g.initial b ∧ g.sameAddress a b
81
82/-- Read-source choices, not a synchronizes-with relation. -/
83def rf (g : Graph n) (a b : Fin n) : Prop := g.read b ∧ g.source b = a
84
85def coherence (g : Graph n) (a b : Fin n) : Prop := g.co a b = true
86
87/-- §8.9.2, with no RMW operations in the language. -/
88def observation (g : Graph n) (a b : Fin n) : Prop :=
89  g.rf a b ∧ g.morallyStrong a b
90
91/-- §8.8: direct release or release followed by a strong write at that address. -/
92def releasePattern (g : Graph n) (a w : Fin n) : Prop :=
93  (a = w ∧ g.release a) ∨
94  (g.release a ∧ g.write w ∧ ¬g.initial w ∧ g.po a w ∧ g.sameAddress a w)
95
96/-- §8.8: direct acquire or strong read followed by acquire at that address. -/
97def acquirePattern (g : Graph n) (r b : Fin n) : Prop :=
98  (r = b ∧ g.acquire b) ∨
99  (g.read r ∧ g.acquire b ∧ g.po r b ∧ g.sameAddress r b)
100
101/-- §8.9.4: pattern endpoints must also be morally strong. -/
102def sync (g : Graph n) (a b : Fin n) : Prop :=
103  (g.event a).thread ≠ (g.event b).thread ∧
104    ∃ w r, g.releasePattern a w ∧ g.acquirePattern r b ∧
105      g.observation w r ∧ g.morallyStrong a b
106
107def baseEdge (g : Graph n) (a b : Fin n) : Prop := g.po a b ∨ g.sync a b
108
109def base (g : Graph n) : Fin n → Fin n → Prop := Path g.baseEdge
110
111/-- §8.9.5, identity addressing and generic proxy: preserve same-address base. -/
112def proxyBase (g : Graph n) (a b : Fin n) : Prop := g.base a b ∧ g.sameAddress a b
113
114/-- Do NOT transitively close this relation: §8.9.5's second clause matters. -/
115def cause (g : Graph n) (a b : Fin n) : Prop :=
116  g.proxyBase a b ∨ ∃ z, g.observation a z ∧ g.proxyBase z b
117
118/-- §8.9.7: RF, coherence, and from-read. -/
119def communication (g : Graph n) (a b : Fin n) : Prop :=
120  g.rf a b ∨ g.coherence a b ∨
121    (g.read a ∧ g.coherence (g.source a) b)
122
123/-- §8.10.5: overlapping PO only; not a global SC union. -/
124def locationEdge (g : Graph n) (a b : Fin n) : Prop :=
125  (g.po a b ∧ g.sameAddress a b) ∨
126  (g.communication a b ∧ g.morallyStrong a b)

128structure Sources (g : Graph n) : Prop where
129  compatible : ∀ r, g.read r → g.write (g.source r) ∧
130    g.sameAddress (g.source r) r ∧
131    (g.event (g.source r)).effect.value = (g.event r).effect.value
132
133/-- All program accesses at a common word are strong and mutually in scope. -/
134structure Coherent (g : Graph n) : Prop where
135  typed : ∀ a b, g.coherence a b → g.write a ∧ g.write b ∧ g.sameAddress a b
136  irrefl : ∀ a, ¬g.coherence a a
137  trans : ∀ a b c, g.coherence a b → g.coherence b c → g.coherence a c
138  total : ∀ a b, g.write a → g.write b → g.sameAddress a b → a ≠ b →
139    g.coherence a b ∨ g.coherence b a
140  initFirst : ∀ a b, g.initial a → g.write b → g.sameAddress a b → a ≠ b →
141    g.coherence a b
142
143/-- Memory constraints, never an assumed publication/output property.
144Fence-SC and RMW axioms have no instances in this typed fragment. Whole-word
145sources encode single-copy atomicity; literal stores provide value grounding.
146Initialization normalization and these restrictions are explained in the ledger. -/
147structure Valid (g : Graph n) : Prop where
148  sources : g.Sources
149  co : g.Coherent
150  base_irrefl : ∀ a, ¬g.base a a
151  coherence_cause : ∀ a b, g.write a → g.write b → g.sameAddress a b →
152    g.cause a b → g.coherence a b
153  no_future : ∀ r w, g.read r → g.write w → g.sameAddress r w →
154    g.cause r w → g.source r ≠ w
155  no_stale : ∀ w r, g.write w → g.read r → g.sameAddress w r →
156    g.cause w r → ¬g.coherence (g.source r) w
157  sc_per_location : ∀ a, ¬Path g.locationEdge a a

159/-- Uniform per-byte lifting: every byte in a read uses its one word source. -/
160def byteSource (g : Graph n) (r : Fin n) (_offset : Fin 4) : Fin n := g.source r

162theorem single_copy (g : Graph n) (r : Fin n) (i j : Fin 4) :
163    g.byteSource r i = g.byteSource r j := rfl

165theorem no_source_coherence_predecessor {g : Graph n} {r w : Fin n} {i j : Fin 4} (h : g.Valid)
166    (same : g.byteSource r i = w) : ¬g.coherence (g.byteSource r j) w := by
167  simpa [byteSource, ← same] using h.co.irrefl (g.source r)

169/-- A reusable sound certificate for a finite witness. `upper` is only a proof
170upper bound on derived base paths; it is not supplied as the actual semantics. -/
171def upperCause (g : Graph n) (upper : Fin n → Fin n → Prop) (a b : Fin n) : Prop :=
172  (upper a b ∧ g.sameAddress a b) ∨
173    ∃ z, g.observation a z ∧ upper z b ∧ g.sameAddress z b

175structure Certificate (g : Graph n) (upper : Fin n → Fin n → Prop)
176    (rank : Fin n → Nat) : Prop where
177  sources : g.Sources
178  co : g.Coherent
179  edge_included : ∀ a b, g.baseEdge a b → upper a b
180  upper_trans : ∀ a b c, upper a b → upper b c → upper a c
181  upper_irrefl : ∀ a, ¬upper a a
182  coherence_cause : ∀ a b, g.write a → g.write b → g.sameAddress a b →
183    g.upperCause upper a b → g.coherence a b
184  no_future : ∀ r w, g.read r → g.write w → g.sameAddress r w →
185    g.upperCause upper r w → g.source r ≠ w
186  no_stale : ∀ w r, g.write w → g.read r → g.sameAddress w r →
187    g.upperCause upper w r → ¬g.coherence (g.source r) w
188  location_rank : ∀ a b, g.locationEdge a b → rank a < rank b
189
190theorem valid_of_certificate {g : Graph n} {upper : Fin n → Fin n → Prop}
191    {rank : Fin n → Nat} (h : g.Certificate upper rank) : g.Valid := by
192  have bound : ∀ a b, g.base a b → upper a b :=
193    fun _ _ hp => Path.contained h.edge_included h.upper_trans hp
194  have causeBound : ∀ a b, g.cause a b → g.upperCause upper a b := by
195    intro a b hc
196    rcases hc with ⟨hb, ha⟩ | ⟨z, ho, hb, ha⟩
197    · exact Or.inl ⟨bound _ _ hb, ha⟩
198    · exact Or.inr ⟨z, ho, bound _ _ hb, ha⟩
199  exact {
200    sources := h.sources
201    co := h.co
202    base_irrefl := fun a hp => h.upper_irrefl a (bound _ _ hp)
203    coherence_cause := fun a b hw hw' ha hc =>
204      h.coherence_cause a b hw hw' ha (causeBound _ _ hc)
205    no_future := fun r w hr hw ha hc => h.no_future r w hr hw ha (causeBound _ _ hc)
206    no_stale := fun w r hw hr ha hc => h.no_stale w r hw hr ha (causeBound _ _ hc)
207    sc_per_location := fun a hp => Nat.lt_irrefl _ (Path.rank_increases rank h.location_rank hp)
208  }
209
210end Graph
