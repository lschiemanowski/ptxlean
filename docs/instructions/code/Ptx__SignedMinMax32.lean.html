22def compute (op : Operation) (a b : Word) : Word :=
23  match op with
24  | .min => if a.toInt < b.toInt then a else b
25  | .max => if a.toInt > b.toInt then a else b

30def family : Pure32.Family Operation :=
31  Pure32.Family.ofFunction (fun _ => 2) (fun _ => 0)
32    (fun op words _ => compute op (words 0) (words 1))
33    (fun target _ => SupportedTarget target)

35theorem compute_toInt (op : Operation) (a b : Word) :
36    (compute op a b).toInt = match op with
37      | .min => min a.toInt b.toInt
38      | .max => max a.toInt b.toInt := by
39  cases op
40  · by_cases h : a.toInt < b.toInt <;> simp [compute, min, h] <;> omega
41  · by_cases h : b.toInt < a.toInt <;> simp [compute, max, h] <;> omega

43theorem results_iff (op : Operation) (words : Fin 2 → Word) (predicates : Fin 0 → Bool) (value : Word) :
44    family.Results op words predicates value ↔ value = compute op (words 0) (words 1) := Iff.rfl

46def lower (i : Instr) : Pure32.Instr family where
47  guard := i.guard
48  operation := i.operation
49  destination := i.destination
50  words := fun j => if j.val = 0 then i.left else i.right
51  predicates := Fin.elim0

53def result (i : Instr) (s : Scalar.State) : Word :=
54  compute i.operation (i.left.eval s) (i.right.eval s)

82theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
83    (enabled : i.guard.eval s = true) :
84    Eval i s next event ↔
85      next = Pure32.write s i.destination (result i s) ∧ event = occurrence s i true := by
86  change Pure32.Eval (lower i) s next event ↔ _
87  rw [Pure32.eval_true_iff (lower i) s next event enabled]
88  constructor
89  · rintro ⟨value, hres, hnext, hevent⟩
90    have hv : value = result i s := by
91      simpa [family, lower, Pure32.Family.ofFunction, Pure32.Instr.wordValues,
92        Pure32.Instr.predicateValues, result, Scalar.Operand32.eval] using hres
93    subst value
94    exact ⟨hnext, hevent⟩
95  · rintro ⟨hnext, hevent⟩
96    refine ⟨result i s, ?_, hnext, hevent⟩
97    simp [family, lower, Pure32.Family.ofFunction, Pure32.Instr.wordValues,
98      result, Scalar.Operand32.eval]

136theorem step_origin (target : Ptx.Target) (program : List Instr)
137    (s next : Scalar.State) (event : Occurrence) (stepped : Step target program s next event) :
138    SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧
139      event.pc = s.pc ∧ event.instruction = lower i ∧ Eval i s next event := by
140  obtain ⟨j, fetch, supported, pc, ins, evaluated⟩ := Pure32.step_origin stepped
141  have original : ∃ i, program[s.pc]? = some i ∧ lower i = j := by
142    cases hx : program[s.pc]? with
143    | none => simp [hx] at fetch
144    | some i =>
145      have hm : some (lower i) = some j := by simpa [hx] using fetch
146      exact ⟨i, by simp, Option.some.inj hm⟩
147  obtain ⟨i, fetchi, eq⟩ := original
148  have ev : Eval i s next event :=
149    Eq.mp (congrArg (fun x : Pure32.Instr family => Pure32.Eval x s next event) eq.symm) evaluated
150  exact ⟨supported, i, fetchi, pc, ins.trans eq.symm, ev⟩

152theorem step_exists (target : Ptx.Target) (program : List Instr) (s : Scalar.State) (i : Instr)
153    (supported : SupportedTarget target) (fetch : program[s.pc]? = some i) :
154    ∃ next event, Step target program s next event := by
155  have mapped : (program.map lower)[s.pc]? = some (lower i) := by
156    simp [fetch]
157  apply Pure32.step_exists (target := target) (program := program.map lower) (i := lower i)
158    (s := s) (fetch := mapped)

175def supportedMnemonic : String → Bool
176  | "min.s32" | "max.s32" => true
177  | _ => false

179def mnemonic : Operation → String
180  | .min => "min.s32"
181  | .max => "max.s32"

183def encode (i : Instr) : Scalar.Text.Statement :=
184  ⟨i.guard, mnemonic i.operation,
185    [.word (.reg i.destination), .word i.left, .word i.right]⟩

187def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
188  match statement.mnemonic, statement.operands with
189  | "min.s32", [.word (.reg d), .word a, .word b] => .ok ⟨statement.guard, .min, d, a, b⟩
190  | "max.s32", [.word (.reg d), .word a, .word b] => .ok ⟨statement.guard, .max, d, a, b⟩
191  | mnemonic, _ => if supportedMnemonic mnemonic then .error (.invalidOperands mnemonic)
192      else .error (.unsupportedMnemonic mnemonic)

194theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
195  cases i with
196  | mk guard operation destination left right => cases operation <;> rfl

198theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) :
199    decode statement = .ok i ↔ statement = encode i := by
200  constructor
201  · intro h
202    rcases i with ⟨g, op, d, a, b⟩
203    cases op <;> cases statement with
204      | mk guard spelling operands =>
205        unfold decode at h
206        split at h
207        all_goals
208          first
209          | (have hv := Except.ok.inj h; cases hv <;> all_goals simp_all [encode, mnemonic])
210          | (split at h <;> simp_all [supportedMnemonic])
211  · intro h
212    subst statement
213    exact decode_encode i
