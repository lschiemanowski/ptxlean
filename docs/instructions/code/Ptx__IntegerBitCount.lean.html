3/-! PTX 9.4 §§9.7.1.15–.16: `popc.b32` and `clz.b32` were introduced
4in PTX 2.0 and require sm_20 or newer. Both accept every b32 value and write
5a u32 result. Only these b32 spellings are modeled here; the ISA's b64 forms
6and other widths/modifiers are not. The package's u32 result name is not a
7mnemonic spelling. The separate memory eligibility mechanism does not check
8these arithmetic forms.
9-/

78theorem popc_zero_iff (a : Word) : UnaryOp.eval .popc a = 0 ↔ a = 0 := by
79  constructor
80  · intro h
81    apply BitVec.eq_of_getLsbD_eq
82    intro i hi
83    have hn := popc_toNat a
84    rw [h] at hn
85    have hn : ((List.range 32).filter (fun j => a.getLsbD j)).length = 0 := by simpa using hn.symm
86    have hmem : i ∈ List.range 32 := List.mem_range.mpr hi
87    have hfalse : a.getLsbD i = false := by
88      cases hv : a.getLsbD i
89      · rfl
90      · have hm : i ∈ (List.range 32).filter (fun j => a.getLsbD j) :=
91          List.mem_filter.mpr ⟨hmem, hv⟩
92        have hne : (List.range 32).filter (fun j => a.getLsbD j) ≠ [] := by
93          intro heq
94          have : i ∈ ([] : List Nat) := heq ▸ hm
95          simp at this
96        have hpos : 0 < ((List.range 32).filter (fun j => a.getLsbD j)).length :=
97          List.length_pos_iff.mpr hne
98        omega
99    simpa using hfalse
100  · rintro rfl
101    change BitVec.ofNat 32 ((List.range 32).filter (fun _ => false)).length = 0
102    rfl

104theorem clz_width_iff (a : Word) : UnaryOp.eval .clz a = 32 ↔ a = 0 := by
105  rw [← BitVec.toNat_inj]
106  rw [clz_toNat]
107  constructor
108  · intro h
109    have hlen : ((List.range 32).reverse.takeWhile (fun i => !a.getLsbD i)).length = 32 := by
110      simpa [BitVec.toNat_ofNat] using h
111    have hall := (takeWhile_length_eq_iff (fun i => !a.getLsbD i)
112      (List.range 32).reverse).mp (by simpa using hlen)
113    apply BitVec.eq_of_getLsbD_eq
114    intro i hi
115    have himem : i ∈ (List.range 32).reverse := List.mem_reverse.mpr (List.mem_range.mpr hi)
116    have hz := hall i himem
117    cases hv : a.getLsbD i <;> simp_all
118  · intro hz
119    subst a
120    have hall : ∀ i ∈ (List.range 32).reverse,
121        Bool.not ((0#32).getLsbD i) = true := by
122      intro i hi
123      have hz : (0#32).getLsbD i = false := by simp
124      simp [hz]
125    have hlen := (takeWhile_length_eq_iff (fun i => Bool.not ((0#32).getLsbD i))
126      (List.range 32).reverse).mpr hall
127    simpa using hlen
