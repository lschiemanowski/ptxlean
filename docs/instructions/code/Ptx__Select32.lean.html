27def compute (a b : Word) (c : Bool) : Word := if c then a else b

29def family : Pure32.Family Operation :=
30  Pure32.Family.ofFunction (fun _ => 2) (fun _ => 1)
31    (fun _ words predicates => compute (words w0) (words w1) (predicates p0))
32    (fun target _ => SupportedTarget target)

34def lower (i : Instr) : Pure32.Instr family :=
35  ⟨i.guard, .selp, i.destination,
36    fun index => if index.val = 0 then i.left else i.right,
37    fun _ => .reg i.predicate true⟩

39def result (i : Instr) (s : Scalar.State) : Word :=
40  compute (i.left.eval s) (i.right.eval s) (s.preds i.predicate)

67theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
68    (enabled : i.guard.eval s = true) :
69    Eval i s next event ↔
70      next = Pure32.write s i.destination (result i s) ∧ event = occurrence s i true := by
71  change Pure32.Eval (lower i) s next event ↔ _
72  rw [Pure32.eval_true_iff (lower i) s next event (by simpa [lower] using enabled)]
73  constructor
74  · rintro ⟨value, h, hn, he⟩
75    have hv : value = result i s := by
76      simpa [Pure32.Instr.wordValues, Pure32.Instr.predicateValues, lower, result, w0, w1, p0,
77        family, Pure32.Family.ofFunction] using h
78    subst value
79    exact ⟨by simpa [lower, family, Pure32.Family.ofFunction,
80      Pure32.Instr.wordValues, Pure32.Instr.predicateValues, w0, w1, p0, result] using hn, he⟩
81  · rintro ⟨hn, he⟩
82    refine ⟨result i s, ?_, hn, he⟩
83    simpa [Pure32.Instr.wordValues, Pure32.Instr.predicateValues, lower, result, w0, w1, p0,
84      family, Pure32.Family.ofFunction]

126theorem step_origin (target : Ptx.Target) (program : List Instr)
127    (s next : Scalar.State) (event : Occurrence) (stepped : Step target program s next event) :
128    SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧
129      event.pc = s.pc ∧ event.instruction = lower i ∧ Eval i s next event := by
130  change Pure32.Step target (program.map lower) s next event at stepped
131  obtain ⟨j, fetch, supported, evaluated⟩ := Pure32.step_origin stepped
132  have hmap : (program[s.pc]?).map lower = some j := by simpa using fetch
133  cases h : program[s.pc]? with
134  | none => simp [h] at hmap
135  | some original =>
136    have hj : lower original = j := Option.some.inj (by simpa [h] using hmap)
137    rcases evaluated with ⟨pc, ins, ev⟩
138    have ho : j.operation = .selp := by cases j.operation <;> rfl
139    change SupportedTarget target at supported
140    refine ⟨supported, original, ?_, pc, ?_, ?_⟩
141    · simpa [h]
142    · simpa [hj] using ins
143    · cases hj
144      exact ev

146theorem step_exists (target : Ptx.Target) (program : List Instr) (s : Scalar.State) (i : Instr)
147    (supported : SupportedTarget target) (fetch : program[s.pc]? = some i) :
148    ∃ next event, Step target program s next event := by
149  have fetch' : (program.map lower)[s.pc]? = some (lower i) := by
150    simpa using congrArg (Option.map lower) fetch
151  obtain ⟨next, event, h⟩ := Pure32.step_exists (lower i) fetch' (by
152    change SupportedTarget target
153    exact supported)
154  exact ⟨next, event, h⟩

176def encode (i : Instr) : Scalar.Text.Statement :=
177  ⟨i.guard, "selp.b32", [.word (.reg i.destination), .word i.left,
178    .word i.right, .predicate i.predicate]⟩

180def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
181  if supportedMnemonic statement.mnemonic then
182    match statement.operands with
183    | [.word (.reg destination), .word left, .word right, .predicate predicate] =>
184      .ok ⟨statement.guard, destination, left, right, predicate⟩
185    | _ => .error (.invalidOperands statement.mnemonic)
186  else .error (.unsupportedMnemonic statement.mnemonic)
187
188theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
189  simp [decode, encode, supportedMnemonic]

191theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) :
192    decode statement = .ok i ↔ statement = encode i := by
193  constructor
194  · intro h
195    rcases statement with ⟨guard, mnemonic, operands⟩
196    simp only [decode] at h
197    by_cases hm : mnemonic = "selp.b32"
198    · subst mnemonic
199      simp [supportedMnemonic] at h
200      split at h
201      · injection h with hi
202        cases hi
203        rfl
204      · simp at h
205    · simp [supportedMnemonic, hm] at h
206  · intro h
207    subst statement
208    exact decode_encode i
