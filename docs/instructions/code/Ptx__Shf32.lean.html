25def compute (op : Operation) (a b c : Word) : Word :=
26  let n := match op with
27    | .leftClamp | .rightClamp => min c.toNat 32
28    | .leftWrap | .rightWrap => c.toNat % 32
29  match op with
30  | .leftClamp | .leftWrap => (b <<< n) ||| (a >>> (32 - n))
31  | .rightClamp | .rightWrap => (b <<< (32 - n)) ||| (a >>> n)

33def family : Pure32.Family Operation :=
34  Pure32.Family.ofFunction (fun _ => 3) (fun _ => 0)
35    (fun op words _ => compute op (words w0) (words w1) (words w2))
36    (fun target _ => SupportedTarget target)

38def lower (i : Instr) : Pure32.Instr family where
39  guard := i.guard
40  operation := i.operation
41  destination := i.destination
42  words := fun j => if j.val = 0 then i.first else if j.val = 1 then i.second else i.control
43  predicates := Fin.elim0

45def result (i : Instr) (s : Scalar.State) : Word :=
46  compute i.operation (i.first.eval s) (i.second.eval s) (i.control.eval s)

67theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
68    (enabled : i.guard.eval s = true) :
69    Eval i s next event ↔
70      next = Pure32.write s i.destination (result i s) ∧ event = occurrence s i true := by
71  change Pure32.Eval (lower i) s next event ↔ _
72  rw [Pure32.eval_true_iff (lower i) s next event enabled]
73  constructor
74  · rintro ⟨value, hres, hnext, hevent⟩
75    have hv : value = result i s := by
76      simpa [family, lower, Pure32.Family.ofFunction, Pure32.Instr.wordValues,
77        Pure32.Instr.predicateValues, result, w0, w1, w2] using hres
78    subst value
79    exact ⟨hnext, hevent⟩
80  · rintro ⟨hnext, hevent⟩
81    refine ⟨result i s, ?_, hnext, hevent⟩
82    simp [family, lower, Pure32.Family.ofFunction, Pure32.Instr.wordValues,
83      result, w0, w1, w2]

121theorem step_origin (target : Ptx.Target) (program : List Instr)
122    (s next : Scalar.State) (event : Occurrence) (stepped : Step target program s next event) :
123    SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧
124      event.pc = s.pc ∧ event.instruction = lower i ∧ Eval i s next event := by
125  obtain ⟨j, fetch, supported, pc, ins, evaluated⟩ := Pure32.step_origin stepped
126  have original : ∃ i, program[s.pc]? = some i ∧ lower i = j := by
127    cases hx : program[s.pc]? with
128    | none => simp [hx] at fetch
129    | some i =>
130      have hm : some (lower i) = some j := by simpa [hx] using fetch
131      exact ⟨i, by simp, Option.some.inj hm⟩
132  obtain ⟨i, fetchi, eq⟩ := original
133  have ev : Eval i s next event :=
134    Eq.mp (congrArg (fun x : Pure32.Instr family => Pure32.Eval x s next event) eq.symm) evaluated
135  exact ⟨supported, i, fetchi, pc, ins.trans eq.symm, ev⟩

170def mnemonic : Operation → String

176def supportedMnemonic (spelling : String) : Bool :=
177  spelling == "shf.l.clamp.b32" || spelling == "shf.l.wrap.b32" ||
178  spelling == "shf.r.clamp.b32" || spelling == "shf.r.wrap.b32"

180def encode (i : Instr) : Scalar.Text.Statement :=
181  ⟨i.guard, mnemonic i.operation,
182    [.word (.reg i.destination), .word i.first, .word i.second, .word i.control]⟩

184def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
185  match statement.mnemonic, statement.operands with
186  | "shf.l.clamp.b32", [.word (.reg d), .word a, .word b, .word c] => .ok ⟨statement.guard, .leftClamp, d, a, b, c⟩
187  | "shf.l.wrap.b32", [.word (.reg d), .word a, .word b, .word c] => .ok ⟨statement.guard, .leftWrap, d, a, b, c⟩
188  | "shf.r.clamp.b32", [.word (.reg d), .word a, .word b, .word c] => .ok ⟨statement.guard, .rightClamp, d, a, b, c⟩
189  | "shf.r.wrap.b32", [.word (.reg d), .word a, .word b, .word c] => .ok ⟨statement.guard, .rightWrap, d, a, b, c⟩
190  | spelling, _ => if supportedMnemonic spelling then .error (.invalidOperands spelling)
191      else .error (.unsupportedMnemonic spelling)

193theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
194  cases i with
195  | mk guard operation destination first second control => cases operation <;> rfl

197theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) :
198    decode statement = .ok i ↔ statement = encode i := by
199  constructor
200  · intro h
201    rcases i with ⟨g, op, d, a, b, c⟩
202    cases op <;> cases statement with
203      | mk guard spelling operands =>
204        unfold decode at h
205        split at h
206        all_goals first
207          | (have hv := Except.ok.inj h; cases hv <;> all_goals simp_all [encode, mnemonic])
208          | (split at h <;> simp_all [supportedMnemonic])
209  · intro h
210    subst statement
211    exact decode_encode i
