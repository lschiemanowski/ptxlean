31def encode : Instruction → String × List Token
32  | .store a s => ("st.relaxed.cta.shared.u32", [.sharedAddressRegister a, .wordRegister s])
33  | .sync r => ("bar.sync", [.resource r])
34  | .load d a => ("ld.relaxed.cta.shared.u32", [.wordRegister d, .sharedAddressRegister a])
35  | .exit => ("exit", [])

37def decode : String → List Token → Option Instruction
38  | "st.relaxed.cta.shared.u32", [.sharedAddressRegister a, .wordRegister s] => some (.store a s)
39  | "bar.sync", [.resource r] => some (.sync r)
40  | "ld.relaxed.cta.shared.u32", [.wordRegister d, .sharedAddressRegister a] => some (.load d a)
41  | "exit", [] => some .exit
42  | _, _ => none

63/-- Shared addresses are 32-bit byte offsets in this CTA's initialized arena. -/
64def AccessSafe (n : Nat) (address : Word) : Prop :=
65  address.toNat % 4 = 0 ∧ address.toNat / 4 < n
66
67def accessIndex (n : Nat) (address : Word) : Option (Fin n) :=
68  if _aligned : address.toNat % 4 = 0 then
69    if bound : address.toNat / 4 < n then some ⟨address.toNat / 4, bound⟩ else none
70  else none
71
72theorem accessIndex_some (h : accessIndex n address = some index) :
73    AccessSafe n address ∧ index.val = address.toNat / 4 := by
74  unfold accessIndex at h
75  split at h
76  · split at h
77    · cases h
78      exact ⟨⟨by assumption, by assumption⟩, rfl⟩
79    · contradiction
80  · contradiction

82def update (values : α → β) [DecidableEq α] (index : α) (value : β) : α → β :=
83  fun other => if other = index then value else values other

124def advance (thread : ThreadState) : ThreadState := {thread with pc := thread.pc+1}

126/-- The caller's fetch selects this operation. Barriers park at their own PC;
127on release all participants advance. Reachability establishes they are all at
128that same barrier, rather than assuming it as a release premise. -/
129def dispatch (config : Config n) (override : Option Word) (thread : Fin n)
130    (instruction : Instruction) (state : State n) : Result n :=
131  let current := state.threads thread
132  match instruction with
133  | .store addressRegister sourceRegister =>
134    if state.arena.owner = config.cta then
135      match accessIndex n (current.addresses addressRegister) with
136      | none => ⟨state, .invalidAddress, []⟩
137      | some index =>
138        ⟨{state with
139            threads := update state.threads thread (advance current)
140            arena := {state.arena with words := update state.arena.words index (current.registers sourceRegister)}},
141         .advanced,
142         [.memory ⟨.store, config.cta, thread, current.pc,
143           current.addresses addressRegister, current.registers sourceRegister⟩]⟩
144    else ⟨state, .wrongOwner, []⟩
145  | .load destinationRegister addressRegister =>
146    if state.arena.owner = config.cta then
147      match accessIndex n (current.addresses addressRegister) with
148      | none => ⟨state, .invalidAddress, []⟩
149      | some index =>
150        let value := override.getD (state.arena.words index)
151        ⟨{state with threads := update state.threads thread {current with pc := current.pc+1, registers := update current.registers destinationRegister value}},
152         .advanced,
153         [.memory ⟨.load, config.cta, thread, current.pc, current.addresses addressRegister, value⟩]⟩
154    else ⟨state, .wrongOwner, []⟩
155  | .sync resource =>
156    let result := Barrier.step (barrierConfig config) state.barrier (barrierRequest config state thread resource)
157    match result.disposition with
158    | .rejectedWrongKey => ⟨state, .wrongBarrierKey, []⟩
159    | .waiting =>
160      ⟨{state with barrier := result.state}, .waiting, result.events.map Event.barrier⟩
161    | .released =>
162      ⟨{state with threads := fun other => advance (state.threads other), barrier := result.state},
163        .released, result.events.map Event.barrier⟩
164  | .exit =>
165    ⟨{state with threads := update state.threads thread {current with halted := true}},
166      .halted, [.exited thread current.pc]⟩

168instance (state : Barrier.State n) (thread : Fin n) : Decidable (Barrier.Waiting state thread) :=
169  inferInstanceAs (Decidable (state.arrived thread = true))

171/-- Waiting and halted threads cannot fetch a later instruction. -/
172def stepWith (config : Config n) (override : Option Word) (thread : Fin n) (state : State n) : Result n :=
173  if (state.threads thread).halted = true then ⟨state, .halted, []⟩
174  else if Barrier.Waiting state.barrier thread then ⟨state, .waiting, []⟩
175  else match program[(state.threads thread).pc]? with
176    | none => ⟨state, .invalidPC, []⟩
177    | some instruction => dispatch config override thread instruction state
178
179def step (config : Config n) (thread : Fin n) (state : State n) : Result n :=
180  stepWith config none thread state

192theorem fetched_step (config : Config n) (override : Option Word) (thread : Fin n) (state : State n)
193    (live : (state.threads thread).halted = false) (runnable : ¬Barrier.Waiting state.barrier thread)
194    (fetch : program[(state.threads thread).pc]? = some instruction) :
195    stepWith config override thread state = dispatch config override thread instruction state := by
196  simp [stepWith, live, runnable, fetch]

198/-- Every emitted effect comes from an actually fetched instruction at a live,
199unblocked thread. The individual dispatch branches determine its full fields. -/
200theorem event_origin (config : Config n) (override : Option Word) (thread : Fin n) (state : State n)
201    (member : event ∈ (stepWith config override thread state).events) :
202    ∃ instruction, (state.threads thread).halted = false ∧
203      ¬Barrier.Waiting state.barrier thread ∧
204      program[(state.threads thread).pc]? = some instruction ∧
205      event ∈ (dispatch config override thread instruction state).events := by
206  unfold stepWith at member
207  split at member
208  · simp at member
209  · rename_i live
210    split at member
211    · simp at member
212    · rename_i runnable
213      split at member
214      · simp at member
215      · rename_i instruction fetch
216        exact ⟨instruction, by simpa using live, runnable, fetch, member⟩

218/-- No dispatch event is discarded by the outer fetched-step wrapper. -/
219theorem event_complete (config : Config n) (override : Option Word) (thread : Fin n) (state : State n)
220    (live : (state.threads thread).halted = false) (runnable : ¬Barrier.Waiting state.barrier thread)
221    (fetch : program[(state.threads thread).pc]? = some instruction)
222    (member : event ∈ (dispatch config override thread instruction state).events) :
223    event ∈ (stepWith config override thread state).events := by
224  rw [fetched_step config override thread state live runnable fetch]
225  exact member

227/-- Emitted shared accesses always passed ownership, alignment and full-word bounds. -/
228theorem dispatch_memory_safe (config : Config n) (override : Option Word) (thread : Fin n)
229    (instruction : Instruction) (state : State n) (effect : MemoryEvent n)
230    (member : Event.memory effect ∈ (dispatch config override thread instruction state).events) :
231    state.arena.owner = config.cta ∧ AccessSafe n effect.address ∧ effect.cta = config.cta ∧
232      effect.thread = thread ∧ effect.pc = (state.threads thread).pc := by
233  cases instruction with
234  | store a s =>
235    simp only [dispatch] at member
236    split at member
237    · rename_i owner
238      split at member
239      · simp at member
240      · rename_i index indexed
241        simp only [List.mem_singleton, Event.memory.injEq] at member
242        subst effect
243        exact ⟨owner, (accessIndex_some indexed).1, rfl, rfl, rfl⟩
244    · simp at member
245  | load d a =>
246    simp only [dispatch] at member
247    split at member
248    · rename_i owner
249      split at member
250      · simp at member
251      · rename_i index indexed
252        simp only [List.mem_singleton, Event.memory.injEq] at member
253        subst effect
254        exact ⟨owner, (accessIndex_some indexed).1, rfl, rfl, rfl⟩
255    · simp at member
256  | sync resource =>
257    simp only [dispatch] at member
258    split at member <;> simp at member
259  | exit => simp [dispatch] at member

261theorem memory_safe (config : Config n) (override : Option Word) (thread : Fin n) (state : State n)
262    (effect : MemoryEvent n) (member : Event.memory effect ∈ (stepWith config override thread state).events) :
263    state.arena.owner = config.cta ∧ AccessSafe n effect.address ∧ effect.cta = config.cta ∧
264      effect.thread = thread ∧ effect.pc = (state.threads thread).pc := by
265  obtain ⟨instruction, _, _, _, source⟩ := event_origin _ _ _ _ member
266  exact dispatch_memory_safe _ _ _ _ _ _ source

268structure Execution (n : Nat) where
269  state : State n
270  trace : List (Event n)

272def runWith (config : Config n) (reads : Fin n → Option Word) :
273    State n → List (Fin n) → Execution n
274  | state, [] => ⟨state, []⟩
275  | state, thread :: rest =>
276    let first := stepWith config (reads thread) thread state
277    let remaining := runWith config reads first.state rest
278    ⟨remaining.state, first.events ++ remaining.trace⟩
279
280def run (config : Config n) : State n → List (Fin n) → Execution n :=
281  runWith config (fun _ => none)

283inductive Runs (config : Config n) (reads : Fin n → Option Word) :
284    State n → List (Fin n) → Execution n → Prop where
285  | nil : Runs config reads state [] ⟨state, []⟩
286  | cons : Runs config reads (stepWith config (reads thread) thread state).state rest remaining →
287      Runs config reads state (thread :: rest)
288        ⟨remaining.state, (stepWith config (reads thread) thread state).events ++ remaining.trace⟩

290theorem runWith_sound (config : Config n) (reads : Fin n → Option Word)
291    (state : State n) (schedule : List (Fin n)) :
292    Runs config reads state schedule (runWith config reads state schedule) := by
293  induction schedule generalizing state with
294  | nil => exact .nil
295  | cons thread rest ih => exact .cons (ih _)

297theorem runWith_append (config : Config n) (reads : Fin n → Option Word)
298    (state : State n) (first second : List (Fin n)) :
299    runWith config reads state (first ++ second) =
300      let left := runWith config reads state first
301      let right := runWith config reads left.state second
302      ⟨right.state, left.trace ++ right.trace⟩ := by
303  induction first generalizing state with
304  | nil => simp [runWith]
305  | cons thread rest ih => simp [runWith, ih, List.append_assoc]
