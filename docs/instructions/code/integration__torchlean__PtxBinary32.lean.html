5/-! Payload-preserving encoded binary32 and a conservative NaN-result envelope.
6This is a numerical foundation, not fetched PTX instruction semantics. The
7reference is checked software arithmetic; no host floating-point primitive is
8used. Finiteness remains an explicit premise of every real-error result. -/
9namespace Ptx.Binary32
10
11open FloatLib.Floats (ExecFloat)
12open FloatLib.Floats.ExecFloat (Binary)
13open FloatLib.Floats.Formats.BinaryInterchange (Model FloatFormat)
14open TorchLean.Floats.IEEE754.IEEE32Exec
15
16abbrev Value := ExecFloat.Binary 8 23
17
18def decode (word : Word) : Value := Binary.ofBits32 (UInt32.ofBitVec word)
19def encode (value : Value) : Word := (Binary.toBits32 value).toBitVec
20
21@[simp] theorem encode_decode (word : Word) : encode (decode word) = word := rfl
22@[simp] theorem decode_encode (value : Value) : decode (encode value) = value := by
23  simp [decode, encode]
24
25def isNaN (word : Word) : Bool := Binary.isNaN (decode word)
26def isFinite (word : Word) : Bool := Binary.isFinite (decode word)
27
28/-- No exceptional encoding is silently interpreted as real zero by this API. -/
29noncomputable def finiteReal (word : Word) : Option ℝ :=
30  if isFinite word then some (Binary.toModel (decode word)).toReal else none
31
32theorem finiteReal_none (word : Word) (h : isFinite word = false) :
33    finiteReal word = none := by simp [finiteReal, h]
34
35theorem finiteReal_some (word : Word) (h : isFinite word = true) :
36    finiteReal word = some (Binary.toModel (decode word)).toReal := by simp [finiteReal, h]
37
38theorem finiteReal_isSome (word : Word) : (finiteReal word).isSome = isFinite word := by
39  simp [finiteReal]; split <;> simp_all
40
41theorem finite_not_nan (word : Word) (h : isFinite word = true) : isNaN word = false :=
42  Model.isNaN_eq_false_of_isFinite_eq_true (Binary.toModel (decode word)) h
43
44/-- Conservative over-approximation: NaN bit realizability is not asserted. -/
45def Envelope (reference output : Word) : Prop :=
46  if isNaN reference then isNaN output = true else output = reference
47
48instance (reference output : Word) : Decidable (Envelope reference output) := by
49  unfold Envelope; infer_instance
50
51theorem envelope_self (reference : Word) : Envelope reference reference := by
52  unfold Envelope; split <;> simp_all
53
54theorem envelope_exists (reference : Word) : ∃ output, Envelope reference output :=
55  ⟨reference, envelope_self reference⟩
56
57theorem envelope_nan (reference output : Word) (h : isNaN reference = true) :
58    Envelope reference output ↔ isNaN output = true := by simp [Envelope, h]
59
60theorem envelope_non_nan (reference output : Word) (h : isNaN reference = false) :
61    Envelope reference output ↔ output = reference := by simp [Envelope, h]
62
63theorem envelope_finite_eq (reference output : Word) (h : isFinite reference = true)
64    (allowed : Envelope reference output) : output = reference :=
65  (envelope_non_nan reference output (finite_not_nan reference h)).mp allowed
66
67theorem envelope_finite (reference output : Word) (h : isFinite reference = true)
68    (allowed : Envelope reference output) : isFinite output = true := by
69  rw [envelope_finite_eq reference output h allowed]; exact h
70
71/-- Selected reference operations; this enum is not a PTX decoder. -/
72inductive Operation where
73  | add | mul
74  deriving DecidableEq, Repr
75
76def reference (op : Operation) (left right : Word) : Word :=
77  encode (match op with
78    | .add => ExecFloat.add (decode left) (decode right)
79    | .mul => ExecFloat.mul (decode left) (decode right))
80
81def Results (op : Operation) (left right output : Word) : Prop :=
82  Envelope (reference op left right) output
83
84theorem results_exists (op : Operation) (left right : Word) :
85    ∃ output, Results op left right output := envelope_exists _
86
87/-- The real target is used only under explicit finite operand hypotheses. -/
88noncomputable def exact (op : Operation) (left right : Word) : ℝ :=
89  match op with
90  | .add => (Binary.toModel (decode left)).toReal + (Binary.toModel (decode right)).toReal
91  | .mul => (Binary.toModel (decode left)).toReal * (Binary.toModel (decode right)).toReal
92
93theorem reference_round (op : Operation) (left right : Word)
94    (finite : isFinite (reference op left right) = true) :
95    (Binary.toModel (decode (reference op left right))).toReal =
96      fp32Round (exact op left right) := by
97  cases op with
98  | add =>
99    simpa [reference, isFinite, exact] using
100      (toReal_add_eq_fp32Round_of_isFinite (x := decode left) (y := decode right)
101        (by simpa [reference, isFinite] using finite))
102  | mul =>
103    simpa [reference, isFinite, exact] using
104      (toReal_mul_eq_fp32Round_of_isFinite (x := decode left) (y := decode right)
105        (by simpa [reference, isFinite] using finite))
106
107/-- This covers all outputs in the envelope, not only the reference witness.
108The actual encoded result-finiteness obligation remains visible. -/
109theorem results_round (op : Operation) (left right output : Word)
110    (finite : isFinite (reference op left right) = true)
111    (allowed : Results op left right output) :
112    finiteReal output = some (fp32Round (exact op left right)) := by
113  have eq := envelope_finite_eq _ _ finite allowed
114  subst output
115  rw [finiteReal_some _ finite, reference_round op left right finite]
116
117/-- Explicit real operands avoid any interpretation of exceptional inputs as
118zero. Finiteness of the encoded result must be established by the consumer. -/
119theorem results_abs_error (op : Operation) (left right output : Word) (x y : ℝ)
120    (leftReal : finiteReal left = some x) (rightReal : finiteReal right = some y)
121    (finite : isFinite (reference op left right) = true)
122    (allowed : Results op left right output) :
123    ∃ z : ℝ, finiteReal output = some z ∧
124      |z - (match op with | .add => x + y | .mul => x * y)| ≤
125        TorchLean.Floats.eps32 (match op with | .add => x + y | .mul => x * y) := by
126  have leftEq : (Binary.toModel (decode left)).toReal = x := by
127    unfold finiteReal at leftReal
128    split at leftReal <;> simp_all
129  have rightEq : (Binary.toModel (decode right)).toReal = y := by
130    unfold finiteReal at rightReal
131    split at rightReal <;> simp_all
132  refine ⟨fp32Round (exact op left right), results_round op left right output finite allowed, ?_⟩
133  have bound := TorchLean.Floats.FP32.round_abs_error (exact op left right)
134  cases op <;> simpa [exact, leftEq, rightEq, fp32Round, TorchLean.Floats.round32] using bound
135
