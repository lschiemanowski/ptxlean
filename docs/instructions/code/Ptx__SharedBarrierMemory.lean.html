12def laneOf (index : Fin (3 * n)) : Fin n := ⟨index.val / 3, by have := index.isLt; omega⟩
13def slotOf (index : Fin (3 * n)) : Fin 3 := ⟨index.val % 3, Nat.mod_lt _ (by decide)⟩
14def indexOf (lane : Fin n) (slot : Fin 3) : Fin (3 * n) :=
15  ⟨3 * lane.val + slot.val, by have := lane.isLt; have := slot.isLt; omega⟩

17@[simp] theorem lane_index (lane : Fin n) (slot : Fin 3) : laneOf (indexOf lane slot) = lane := by
18  apply Fin.ext; simp [laneOf, indexOf]; omega

20@[simp] theorem slot_index (lane : Fin n) (slot : Fin 3) : slotOf (indexOf lane slot) = slot := by
21  apply Fin.ext; simp [slotOf, indexOf]

23theorem index_eta (index : Fin (3*n)) : indexOf (laneOf index) (slotOf index) = index := by
24  apply Fin.ext; simp [laneOf, slotOf, indexOf]; omega

26@[simp] theorem index_eq (a b : Fin n) (x y : Fin 3) :
27    indexOf a x = indexOf b y ↔ a = b ∧ x = y := by
28  constructor
29  · intro h
30    exact ⟨by simpa using congrArg laneOf h, by simpa using congrArg slotOf h⟩
31  · rintro ⟨rfl, rfl⟩; rfl

33theorem forall_index_iff (p : Fin (3*n) → Prop) :
34    (∀ index, p index) ↔ ∀ lane slot, p (indexOf lane slot) := by
35  constructor
36  · exact fun h lane slot => h _
37  · intro h index; simpa only [index_eta] using h (laneOf index) (slotOf index)

39theorem exists_index_iff (p : Fin (3*n) → Prop) :
40    (∃ index, p index) ↔ ∃ lane slot, p (indexOf lane slot) := by
41  constructor
42  · rintro ⟨index, h⟩; exact ⟨laneOf index, slotOf index, by simpa only [index_eta] using h⟩
43  · rintro ⟨lane, slot, h⟩; exact ⟨_, h⟩

45def event (old inputs reads : Fin n → Word) (partner : Fin n → Fin n)
46    (lane : Fin n) (slot : Fin 3) : Ptx.Occurrence :=
47  match slot.val with
48  | 0 => ⟨none, 0, ⟨.init, lane.val, old lane⟩⟩
49  | 1 => ⟨some lane.val, 0, ⟨.store .relaxed, lane.val, inputs lane⟩⟩
50  | _ => ⟨some lane.val, 2, ⟨.load .relaxed, (partner lane).val, reads lane⟩⟩

53    (source : Fin (3*n) → Fin (3*n)) (co : Fin (3*n) → Fin (3*n) → Bool) : Graph (3*n) :=
54  ⟨fun i => event old inputs reads partner (laneOf i) (slotOf i), source, co⟩

61/-- Coarse phase levels, not timestamps of a total issue-order execution.
62Only own-thread before/after edges are extracted from these levels. -/
63def phaseOrder (n : Nat) : CollectiveOrder.Certificate (Fin (3*n)) (Option (Fin n)) Unit where
64  owner := fun e => if slotOf e = 0 then none else some (laneOf e)
65  participates := fun _ t => t.isSome = true
66  memoryPos := fun e => if slotOf e = 0 then 0 else if slotOf e = 1 then 1 else 5
67  arrivalPos := fun _ _ => 2
68  completionPos := fun _ => 3
69  resumePos := fun _ _ => 4
70  arrival_before_completion := by intros; decide
71  completion_before_resume := by intros; decide

73def barrierEdge (a b : Fin (3*n)) : Prop := slotOf a = 1 ∧ slotOf b = 2

75theorem cross_iff (a b : Fin (3*n)) : (phaseOrder n).cross a b ↔ barrierEdge a b := by
76  rw [← index_eta a, ← index_eta b]
77  generalize laneOf a = la
78  generalize laneOf b = lb
79  generalize slotOf a = sa
80  generalize slotOf b = sb
81  have casesA : sa = 0 ∨ sa = 1 ∨ sa = 2 := by omega
82  have casesB : sb = 0 ∨ sb = 1 ∨ sb = 2 := by omega
83  rcases casesA with rfl | rfl | rfl <;> rcases casesB with rfl | rfl | rfl
84  all_goals simp [CollectiveOrder.Certificate.cross, CollectiveOrder.Certificate.Before,
85    CollectiveOrder.Certificate.After, phaseOrder, barrierEdge]

87theorem source_choices (old inputs reads : Fin n → Word) (partner source co)
88    (sources : (graph old inputs reads partner source co).Sources) (lane : Fin n) :
89    source (indexOf lane 2) = indexOf (partner lane) 0 ∨
90      source (indexOf lane 2) = indexOf (partner lane) 1 := by
91  have h := sources.compatible (indexOf lane 2) (by simp [Graph.read, event])
92  change (graph old inputs reads partner source co).write (source (indexOf lane 2)) ∧
93    (graph old inputs reads partner source co).sameAddress (source (indexOf lane 2)) (indexOf lane 2) ∧
94    ((graph old inputs reads partner source co).event (source (indexOf lane 2))).effect.value =
95    ((graph old inputs reads partner source co).event (indexOf lane 2)).effect.value at h
96  generalize hi : source (indexOf lane 2) = i at h ⊢
97  rw [← index_eta i] at h ⊢
98  generalize laneOf i = owner at h ⊢
99  generalize slotOf i = slot at h ⊢
100  have casesSlot : slot = 0 ∨ slot = 1 ∨ slot = 2 := by omega
101  rcases casesSlot with rfl | rfl | rfl
102  all_goals simp [Graph.write, Graph.sameAddress, event] at h
103  · exact Or.inl (by have eq : owner = partner lane := Fin.ext h.1; simp [eq])
104  · exact Or.inr (by have eq : owner = partner lane := Fin.ext h.1; simp [eq])

106theorem source_after_barrier (old inputs reads : Fin n → Word) (partner source co)
107    (valid : Graph.Ordered.Valid (graph old inputs reads partner source co) barrierEdge)
108    (lane : Fin n) : source (indexOf lane 2) = indexOf (partner lane) 1 := by
109  let g := graph old inputs reads partner source co
110  have choices := source_choices old inputs reads partner source co valid.sources lane
111  rcases choices with initial | written
112  · have obsolete := valid.co.initFirst (indexOf (partner lane) 0) (indexOf (partner lane) 1)
113      (by simp [Graph.initial, event]) (by simp [Graph.write, event])
114      (by simp [Graph.sameAddress, event]) (by simp)
115    have ordered : Graph.Ordered.cause g barrierEdge (indexOf (partner lane) 1) (indexOf lane 2) :=
116      Graph.Ordered.extra_cause g barrierEdge _ _ (by simp [barrierEdge]) (by simp [g, Graph.sameAddress, event])
117    exact False.elim (valid.no_stale _ _ (by simp [Graph.write, event])
118      (by simp [Graph.read, event]) (by simp [Graph.sameAddress, event]) ordered
119      (by simpa [graph, initial] using obsolete))
120  · exact written

122theorem read_after_barrier (old inputs reads : Fin n → Word) (partner source co)
123    (valid : Graph.Ordered.Valid (graph old inputs reads partner source co) barrierEdge)
124    (lane : Fin n) : reads lane = inputs (partner lane) := by
125  have source := source_after_barrier old inputs reads partner source co valid lane
126  have h := Graph.value_of_source _ valid.sources _ (indexOf lane 2)
127    (by simp [Graph.read, event]) source
128  simpa [event] using h

130def witnessSource (partner : Fin n → Fin n) (fresh : Bool) (i : Fin (3*n)) : Fin (3*n) :=
131  indexOf (partner (laneOf i)) (if fresh then 1 else 0)

133def witnessCo (a b : Fin (3*n)) : Bool :=
134  laneOf a == laneOf b && slotOf a == 0 && slotOf b == 1

136def freshGraph (old inputs : Fin n → Word) (partner : Fin n → Fin n) : Graph (3*n) :=
137  graph old inputs (fun lane => inputs (partner lane)) partner (witnessSource partner true) witnessCo

139def staleGraph (old inputs : Fin n → Word) (partner : Fin n → Fin n) : Graph (3*n) :=
140  graph old inputs (fun lane => old (partner lane)) partner (witnessSource partner false) witnessCo

142def localUpper (a b : Fin (3*n)) : Prop :=
143  laneOf a = laneOf b ∧ slotOf a = 1 ∧ slotOf b = 2

145def staleRank (i : Fin (3*n)) : Nat :=
146  if slotOf i = 0 then 0 else if slotOf i = 1 then 2 else 1

195/-- Source identity, not just coincident equal values, excludes the old graph. -/
196theorem stale_with_barrier_invalid (old inputs : Fin n → Word)
197    (partner : Fin n → Fin n) (lane : Fin n) :
198    ¬Graph.Ordered.Valid (staleGraph old inputs partner) barrierEdge := by
199  intro valid
200  have h := source_after_barrier old inputs (fun lane => old (partner lane)) partner
201    (witnessSource partner false) witnessCo valid lane
202  simp [witnessSource] at h

204theorem fresh_phase_valid (old inputs : Fin n → Word) (partner : Fin n → Fin n) :
205    Graph.Ordered.Valid (freshGraph old inputs partner) (phaseOrder n).cross :=
206  Graph.Ordered.valid_restrict _ (fun a b h => (cross_iff a b).mp h)
207    (fresh_valid old inputs partner)

209theorem read_after_phase (old inputs reads : Fin n → Word) (partner source co)
210    (valid : Graph.Ordered.Valid (graph old inputs reads partner source co) (phaseOrder n).cross)
211    (lane : Fin n) : reads lane = inputs (partner lane) :=
212  read_after_barrier old inputs reads partner source co

215/-- A selected two-warp launch with 32 lanes per warp, not a universal PTX

217def partner64 (lane : Fin 64) : Fin 64 :=
218  ⟨(lane.val + 32) % 64, Nat.mod_lt _ (by decide)⟩

220def warp32 (lane : Fin 64) : Fin 2 :=
