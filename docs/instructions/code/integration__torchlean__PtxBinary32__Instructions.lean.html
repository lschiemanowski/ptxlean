5/-! Explicit `add.rn.f32` and `mul.rn.f32` scalar leaves. Immediate words in
6this context are already-decoded exact `0f`/`0F` binary32 bit literals; no
7integer-to-float conversion or decimal parsing is performed. Register type
8compatibility (`.b32`/`.f32`) remains a frontend obligation. This models the
9conservative `Ptx.Binary32.Results` envelope, not a particular GPU result. -/
10namespace Ptx.Scalar.Binary32

12structure Instr where
13  guard : Scalar.Guard := .always
14  operation : Ptx.Binary32.Operation
15  destination : Nat
16  left : Scalar.Operand32
17  right : Scalar.Operand32
18  deriving DecidableEq, Repr
19
20structure Occurrence where
21  pc : Nat
22  instruction : Instr
23  executed : Bool
24  reads : List Scalar.Register
25  writes : List Scalar.Register
26  memory : Option Scalar.MemoryEffect
27  deriving DecidableEq, Repr
28
29def occurrence (s : Scalar.State) (i : Instr) (executed : Bool) : Occurrence :=
30  ⟨s.pc, i, executed, i.guard.reads ++
31    (if executed then i.left.reads ++ i.right.reads else []),
32    (if executed then [.word i.destination] else []), none⟩
33
34def SupportedTarget (target : Ptx.Target) : Prop :=
35  target.isa = 94 ∧ 20 ≤ target.sm
36
37inductive Eval (i : Instr) : Scalar.State → Scalar.State → Occurrence → Prop
38  | skipped {s} (disabled : i.guard.eval s = false) :
39      Eval i s {s with pc := s.pc + 1} (occurrence s i false)
40  | executed {s value} (enabled : i.guard.eval s = true)
41      (result : Ptx.Binary32.Results i.operation (i.left.eval s) (i.right.eval s) value) :
42      Eval i s {{s with pc := s.pc + 1} with
43        regs := Scalar.update s.regs i.destination value} (occurrence s i true)
44
45def Step (target : Ptx.Target) (program : List Instr)
46    (s next : Scalar.State) (event : Occurrence) : Prop :=
47  SupportedTarget target ∧
48    ∃ instruction, program[s.pc]? = some instruction ∧ Eval instruction s next event

50theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
51    (enabled : i.guard.eval s = true) :
52    Eval i s next event ↔ ∃ value,
53      Ptx.Binary32.Results i.operation (i.left.eval s) (i.right.eval s) value ∧
54      next = {{s with pc := s.pc + 1} with regs := Scalar.update s.regs i.destination value} ∧
55      event = occurrence s i true := by
56  constructor
57  · intro h; cases h with
58    | skipped disabled => simp [enabled] at disabled
59    | executed _ result => exact ⟨_, result, rfl, rfl⟩
60  · rintro ⟨value, result, rfl, rfl⟩; exact .executed enabled result
61
62theorem eval_false_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
63    (disabled : i.guard.eval s = false) :
64    Eval i s next event ↔ next = {s with pc := s.pc + 1} ∧ event = occurrence s i false := by
65  constructor
66  · intro h; cases h with
67    | skipped _ => exact ⟨rfl, rfl⟩
68    | executed enabled _ => simp [disabled] at enabled
69  · rintro ⟨rfl, rfl⟩; exact .skipped disabled

71theorem eval_destination (i : Instr) (s next : Scalar.State) (event : Occurrence)
72    (hEval : Eval i s next event) (enabled : i.guard.eval s = true) :
73    Ptx.Binary32.Results i.operation (i.left.eval s) (i.right.eval s)
74      (next.regs i.destination) := by
75  cases hEval with
76  | skipped disabled => simp [enabled] at disabled
77  | executed _ result => simpa using result

79theorem eval_frame (i : Instr) (s next : Scalar.State) (event : Occurrence)
80    (hEval : Eval i s next event) :
81    next.pc = s.pc + 1 ∧ next.memory = s.memory ∧ next.addrs = s.addrs ∧ next.preds = s.preds := by
82  cases hEval <;> simp_all

84theorem eval_other (i : Instr) (s next : Scalar.State) (event : Occurrence) (other : Nat)
85    (hEval : Eval i s next event) (different : other ≠ i.destination) :
86    next.regs other = s.regs other := by
87  cases hEval with
88  | skipped _ => rfl
89  | executed _ _ => simp [Scalar.update, different]

91theorem eval_event (i : Instr) (s next : Scalar.State) (event : Occurrence)
92    (hEval : Eval i s next event) : event = occurrence s i (i.guard.eval s) := by
93  cases hEval with
94  | skipped h => simp [h]
95  | executed h _ => simp [h]

97theorem eval_exists (i : Instr) (s : Scalar.State) : ∃ next event, Eval i s next event := by
98  by_cases h : i.guard.eval s = true
99  · obtain ⟨value, hv⟩ := Ptx.Binary32.results_exists i.operation (i.left.eval s) (i.right.eval s)
100    exact ⟨{{s with pc := s.pc + 1} with regs := Scalar.update s.regs i.destination value},
101      occurrence s i true, .executed h hv⟩
102  · have hf : i.guard.eval s = false := Bool.eq_false_of_not_eq_true h
103    exact ⟨{s with pc := s.pc + 1}, occurrence s i false, .skipped hf⟩

105theorem step_origin (target : Ptx.Target) (program : List Instr)
106    (s next : Scalar.State) (event : Occurrence)
107    (hStep : Step target program s next event) :
108    SupportedTarget target ∧ ∃ i,
109      program[s.pc]? = some i ∧ event.instruction = i ∧ Eval i s next event := by
110  rcases hStep with ⟨hs, i, fetch, he⟩
111  exact ⟨hs, i, fetch, by cases he <;> rfl, he⟩
112
113theorem step_exists (target : Ptx.Target) (program : List Instr)
114    (s : Scalar.State) (i : Instr) (supported : SupportedTarget target)
115    (fetch : program[s.pc]? = some i) : ∃ next event, Step target program s next event := by
116  obtain ⟨next, event, he⟩ := eval_exists i s
117  exact ⟨next, event, supported, i, fetch, he⟩

119theorem step_no_fetch (target : Ptx.Target) (program : List Instr)
120    (s next : Scalar.State) (event : Occurrence) (missing : program[s.pc]? = none) :
121    ¬ Step target program s next event := by
122  rintro ⟨_, i, fetch, _⟩; rw [missing] at fetch; contradiction

124theorem step_unsupported_target (target : Ptx.Target) (program : List Instr)
125    (s next : Scalar.State) (event : Occurrence) (unsupported : ¬ SupportedTarget target) :
126    ¬ Step target program s next event := by
127  rintro ⟨hs, _⟩; exact unsupported hs

131def mnemonic : Ptx.Binary32.Operation → String
132  | .add => "add.rn.f32"
133  | .mul => "mul.rn.f32"
134
135def supportedMnemonic (name : String) : Bool := name == "add.rn.f32" || name == "mul.rn.f32"
136
137def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
138  let operation := if statement.mnemonic == "add.rn.f32" then some .add
139    else if statement.mnemonic == "mul.rn.f32" then some .mul else none
140  match operation with
141  | none => .error (.unsupportedMnemonic statement.mnemonic)
142  | some op => match statement.operands with
143    | [.word (.reg d), .word a, .word b] => .ok ⟨statement.guard, op, d, a, b⟩
144    | _ => .error (.invalidOperands statement.mnemonic)
145
146def encode (instruction : Instr) : Scalar.Text.Statement :=
147  ⟨instruction.guard, mnemonic instruction.operation,
148    [.word (.reg instruction.destination), .word instruction.left, .word instruction.right]⟩
149
150theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
151  cases i with
152  | mk g op d a b => cases op <;> rfl
153
154theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) :
155    decode statement = .ok i ↔ statement = encode i := by
156  constructor
157  · intro h
158    cases statement with
159    | mk g m ops =>
160      cases i with
161      | mk gi oi di li ri =>
162        by_cases hm : m = "add.rn.f32"
163        · subst m
164          cases oi with
165          | add =>
166            simp [decode] at h
167            split at h
168            · injection h with hInstr
169              cases hInstr
170              rfl
171            · contradiction
172          | mul => simp [decode] at h; split at h <;> simp_all
173        · by_cases hm' : m = "mul.rn.f32"
174          · subst m
175            cases oi with
176            | add => simp [decode] at h; split at h <;> simp_all
177            | mul =>
178              simp [decode] at h
179              split at h
180              · injection h with hInstr
181                cases hInstr
182                rfl
183              · contradiction
184          · simp [decode, hm, hm'] at h
185  · rintro rfl; exact (decode_encode i)
