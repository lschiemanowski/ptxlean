9/-- Polarity matches `Guard.pred`: false selects the negated register value.
10Immediate/polarity forms are generic mechanisms, not promises of PTX text legality. -/
11inductive OperandPred where
12  | reg (index : Nat) (positive : Bool)
13  | imm (value : Bool)
14  deriving DecidableEq, Repr
15
16def OperandPred.eval (s : State) : OperandPred → Bool
17  | .reg index positive => s.preds index == positive
18  | .imm value => value
19
20def OperandPred.reads : OperandPred → List Register
21  | .reg index _ => [.predicate index]
22  | .imm _ => []
23
24@[simp] theorem predicate_positive (s : State) (index : Nat) :
25    OperandPred.eval s (.reg index true) = s.preds index := by simp [OperandPred.eval]
26
27@[simp] theorem predicate_negative (s : State) (index : Nat) :
28    OperandPred.eval s (.reg index false) = !s.preds index := by simp [OperandPred.eval]

30/-- Finite arities and nonempty relations; no determinism or PTX fidelity is
31implied by this structure. Unsupported targets are excluded at fetched steps. -/
32structure Family (Operation : Type) where
33  wordArity : Operation → Nat
34  predicateArity : Operation → Nat
35  Results : (op : Operation) → (Fin (wordArity op) → Word) →
36    (Fin (predicateArity op) → Bool) → Word → Prop
37  inhabited : ∀ op words predicates, ∃ value, Results op words predicates value
38  Supported : Target → Operation → Prop
39
40/-- Package a total computation as a single-result family. The supplied function
41and target predicate still need their own instruction source justification. -/
42def Family.ofFunction (wordArity predicateArity : Operation → Nat)
43    (compute : (op : Operation) → (Fin (wordArity op) → Word) →
44      (Fin (predicateArity op) → Bool) → Word)
45    (supported : Target → Operation → Prop) : Family Operation where
46  wordArity := wordArity
47  predicateArity := predicateArity
48  Results := fun op words predicates value => value = compute op words predicates
49  inhabited := fun op words predicates => ⟨compute op words predicates, rfl⟩
50  Supported := supported
51
52theorem Family.ofFunction_results (wordArity predicateArity : Operation → Nat)
53    (compute : (op : Operation) → (Fin (wordArity op) → Word) →
54      (Fin (predicateArity op) → Bool) → Word)
55    (supported : Target → Operation → Prop) (op : Operation)
56    (words : Fin (wordArity op) → Word) (predicates : Fin (predicateArity op) → Bool) (value : Word) :
57    (Family.ofFunction wordArity predicateArity compute supported).Results op words predicates value ↔
58      value = compute op words predicates := Iff.rfl

62structure Instr (family : Family Operation) where
63  guard : Guard := .always
64  operation : Operation
65  destination : Nat
66  words : Fin (family.wordArity operation) → Operand32
67  predicates : Fin (family.predicateArity operation) → OperandPred
68
69def Instr.wordValues (i : Instr family) (s : State) :
70    Fin (family.wordArity i.operation) → Word := fun index => (i.words index).eval s
71
72def Instr.predicateValues (i : Instr family) (s : State) :
73    Fin (family.predicateArity i.operation) → Bool := fun index => (i.predicates index).eval s

75/-- Ordered syntactic operands, preserving repeats and including every branch's
76inputs in a selecting operation. This is not semantic dependency metadata. -/
77def Instr.sourceReads (i : Instr family) : List Register :=
78  (List.ofFn i.words).flatMap Operand32.reads ++
79    (List.ofFn i.predicates).flatMap OperandPred.reads
80
81theorem Instr.sourceReads_word (i : Instr family) (index : Fin (family.wordArity i.operation))
82    (register : Nat) (source : i.words index = .reg register) :
83    .word register ∈ i.sourceReads := by
84  apply List.mem_append_left
85  apply List.mem_flatMap.mpr
86  exact ⟨i.words index, List.mem_ofFn.mpr ⟨index, rfl⟩, by simp [source, Operand32.reads]⟩
87
88theorem Instr.sourceReads_predicate (i : Instr family)
89    (index : Fin (family.predicateArity i.operation)) (register : Nat) (positive : Bool)
90    (source : i.predicates index = .reg register positive) : .predicate register ∈ i.sourceReads := by
91  apply List.mem_append_right
92  apply List.mem_flatMap.mpr
93  exact ⟨i.predicates index, List.mem_ofFn.mpr ⟨index, rfl⟩,
94    by simp [source, OperandPred.reads]⟩

96structure Occurrence (family : Family Operation) where
97  pc : Nat
98  instruction : Instr family
99  executed : Bool
100  reads : List Register
101  writes : List Register
102  memory : Option MemoryEffect

104def occurrence (s : State) (i : Instr family) (executed : Bool) : Occurrence family :=
105  ⟨s.pc, i, executed, i.guard.reads ++ (if executed then i.sourceReads else []),
106    (if executed then [.word i.destination] else []), none⟩

108def write (s : State) (destination : Nat) (value : Word) : State :=
109  {s with pc := s.pc + 1, regs := update s.regs destination value}

111inductive Eval (i : Instr family) : State → State → Occurrence family → Prop
112  | skipped {s} (disabled : i.guard.eval s = false) :
113      Eval i s {s with pc := s.pc + 1} (occurrence s i false)
114  | executed {s value} (enabled : i.guard.eval s = true)
115      (result : family.Results i.operation (i.wordValues s) (i.predicateValues s) value) :
116      Eval i s (write s i.destination value) (occurrence s i true)

118/-- Successful admission only: unsupported operations have no step even under a
119false guard. This does not specify the hardware's invalid-instruction behavior. -/
120def Step (target : Target) (program : List (Instr family))
121    (s next : State) (event : Occurrence family) : Prop :=
122  ∃ i, program[s.pc]? = some i ∧ family.Supported target i.operation ∧ Eval i s next event

127theorem eval_true_iff (i : Instr family) (s next : State) (event : Occurrence family)
128    (enabled : i.guard.eval s = true) :
129    Eval i s next event ↔ ∃ value,
130      family.Results i.operation (i.wordValues s) (i.predicateValues s) value ∧
131      next = write s i.destination value ∧ event = occurrence s i true := by
132  constructor
133  · intro h
134    cases h with
135    | skipped disabled => simp [enabled] at disabled
136    | executed _ result => exact ⟨_, result, rfl, rfl⟩
137  · rintro ⟨value, result, rfl, rfl⟩
138    exact .executed enabled result

140theorem eval_false_iff (i : Instr family) (s next : State) (event : Occurrence family)
141    (disabled : i.guard.eval s = false) :
142    Eval i s next event ↔ next = {s with pc := s.pc + 1} ∧ event = occurrence s i false := by
143  constructor
144  · intro h
145    cases h with
146    | skipped _ => exact ⟨rfl, rfl⟩
147    | executed enabled _ => simp [disabled] at enabled
148  · rintro ⟨rfl, rfl⟩
149    exact .skipped disabled

151theorem eval_destination (evaluated : Eval i s next event) (enabled : i.guard.eval s = true) :
152    family.Results i.operation (i.wordValues s) (i.predicateValues s) (next.regs i.destination) := by
153  cases evaluated with
154  | skipped disabled => simp [enabled] at disabled

215theorem eval_exists (i : Instr family) (s : State) : ∃ next event, Eval i s next event := by
216  cases enabled : i.guard.eval s with
217  | false => exact ⟨_, _, .skipped enabled⟩
218  | true =>
219    obtain ⟨value, result⟩ := family.inhabited i.operation (i.wordValues s) (i.predicateValues s)
220    exact ⟨_, _, .executed enabled result⟩

222/-- Determinism is conditional; relational families may permit several outputs. -/
223theorem eval_deterministic
224    (functional : ∀ words predicates a b,
225      family.Results i.operation words predicates a →
226      family.Results i.operation words predicates b → a = b)
227    (left : Eval i s next₁ event₁) (right : Eval i s next₂ event₂) :
228    next₁ = next₂ ∧ event₁ = event₂ := by
229  cases left with
230  | skipped disabled =>
231    cases right with
232    | skipped _ => exact ⟨rfl,rfl⟩
233    | executed enabled _ => simp [disabled] at enabled
234  | executed enabled result₁ =>
235    cases right with
236    | skipped disabled => simp [enabled] at disabled
237    | executed _ result₂ =>
238      have same := functional _ _ _ _ result₁ result₂
239      cases same
240      exact ⟨rfl,rfl⟩

242/-- Distinct permitted results stay distinct transitions; the shared mechanism
243does not select one value from a nondeterministic relation. -/
244theorem eval_preserves_choices (enabled : i.guard.eval s = true)
245    (left : family.Results i.operation (i.wordValues s) (i.predicateValues s) a)
246    (right : family.Results i.operation (i.wordValues s) (i.predicateValues s) b)
247    (different : a ≠ b) :
248    Eval i s (write s i.destination a) (occurrence s i true) ∧
249    Eval i s (write s i.destination b) (occurrence s i true) ∧
250    write s i.destination a ≠ write s i.destination b := by
251  refine ⟨.executed enabled left, .executed enabled right, ?_⟩
252  intro equal
253  have same := congrArg (fun state => state.regs i.destination) equal
254  exact different (by simpa [write] using same)

256theorem step_origin (stepped : Step target program s next event) :
257    ∃ i, program[s.pc]? = some i ∧ family.Supported target i.operation ∧
258      event.pc = s.pc ∧ event.instruction = i ∧ Eval i s next event := by
259  obtain ⟨i, fetched, supported, evaluated⟩ := stepped
260  refine ⟨i, fetched, supported, ?_, ?_, evaluated⟩
261  · rw [eval_event evaluated]; rfl
262  · rw [eval_event evaluated]; rfl
263
264theorem step_iff_of_fetch (fetch : program[s.pc]? = some i) :
265    Step target program s next event ↔ family.Supported target i.operation ∧ Eval i s next event := by
266  constructor
267  · rintro ⟨found, fetched, supported, evaluated⟩
268    have same : found = i := Option.some.inj (fetched.symm.trans fetch)
269    subst found
270    exact ⟨supported, evaluated⟩
271  · rintro ⟨supported, evaluated⟩
272    exact ⟨i, fetch, supported, evaluated⟩
273
274theorem step_frame (stepped : Step target program s next event) :
275    next.pc = s.pc + 1 ∧ next.memory = s.memory ∧ next.addrs = s.addrs ∧ next.preds = s.preds := by
276  obtain ⟨_, _, _, evaluated⟩ := stepped
277  exact eval_frame evaluated

279theorem step_false_iff (fetch : program[s.pc]? = some i) (disabled : i.guard.eval s = false) :
280    Step target program s next event ↔ family.Supported target i.operation ∧
281      next = {s with pc := s.pc + 1} ∧ event = occurrence s i false := by

299/-- A record naming another instruction cannot be accepted as this fetched step. -/
300theorem step_wrong_instruction (fetch : program[s.pc]? = some i)
301    (different : event.instruction ≠ i) : ¬ Step target program s next event := by
302  intro stepped
303  have evaluated := ((step_iff_of_fetch fetch).mp stepped).2
304  exact different (by rw [eval_event evaluated]; rfl)
