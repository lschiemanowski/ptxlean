3/-!
4# A scalar machine with explicit execution outcomes
5
6This is a concrete, sequential arena machine for a typed PTX integer subset.
7Candidate reads can be supplied explicitly through `stepWith`/`runWith`; such
8candidates are not automatically permitted PTX weak-memory executions. Registers
9and initial memory are supplied by the caller, never implicitly initialized.

88/-- Shift counts are unsigned 32-bit values. PTX clamps counts at 32. -/
89def BinOp.eval : BinOp → Word → Word → Word
90  | .add, a, b => a + b
91  | .sub, a, b => a - b
92  | .mulLo, a, b => a * b
93  | .and, a, b => a &&& b
94  | .or, a, b => a ||| b
95  | .xor, a, b => a ^^^ b
96  | .shl, a, b => if b.toNat < 32 then a <<< b.toNat else 0
97  | .shr, a, b => if b.toNat < 32 then a >>> b.toNat else 0
98  | .minU, a, b => if a.toNat ≤ b.toNat then a else b
99  | .maxU, a, b => if a.toNat ≥ b.toNat then a else b

101/-- PTX 9.4 `clz.b32` and `popc.b32`; their result register is u32. -/
102def UnaryOp.eval : UnaryOp → Word → Word
103  | .clz, a => BitVec.ofNat 32 ((List.range 32).reverse.takeWhile
104      (fun i => !a.getLsbD i)).length
105  | .popc, a => BitVec.ofNat 32 ((List.range 32).filter
106      (fun i => a.getLsbD i)).length

193/-- Byte address validity in the caller-supplied arena. No wrapping arena extent. -/
194def ValidAddress (memory : List Word) (address : Address) : Prop :=
195  address.toNat % 4 = 0 ∧ address.toNat / 4 < memory.length
196
197instance (memory : List Word) (address : Address) : Decidable (ValidAddress memory address) := by
198  unfold ValidAddress; infer_instance

200/-- A checked byte-to-word lookup retains the exact invalid-access reason. -/
201def addressIndex (memory : List Word) (address : Address) : Except Fault Nat :=
202  if address.toNat % 4 = 0 then
203    if address.toNat / 4 < memory.length then .ok (address.toNat / 4)
204    else .error (.outOfBounds address)
205  else .error (.misaligned address)
206
207theorem addressIndex_ok_iff : addressIndex memory address = .ok index ↔
208    ValidAddress memory address ∧ index = address.toNat / 4 := by
209  by_cases ha : address.toNat % 4 = 0 <;>
210    by_cases hb : address.toNat / 4 < memory.length <;>
211    simp [addressIndex, ValidAddress, ha, hb, eq_comm]

213/-- Execute one fetched instruction. `readOverride` is a candidate value for this
214step's load; `none` selects the concrete current arena value. Both modes check bounds. -/
215def eval (readOverride : Option Word) (instruction : Instr) (s : State) : StepResult :=
216  match instruction.op with
217  | .unsupported spelling => .unsupported spelling
218  | _ =>
219      if instruction.guard.eval s then
220        let advance := {s with pc := s.pc + 1}
221        let event := occurrence s instruction true
222        match instruction.op with
223        | .mov32 destination source =>
224            .next {advance with regs := update s.regs destination (source.eval s)} event
225        | .bin32 operation destination left right =>
226            .next {advance with regs := update s.regs destination (operation.eval (left.eval s) (right.eval s))} event
227        | .unary32 operation destination source =>
228            .next {advance with regs := update s.regs destination (operation.eval (source.eval s))} event
229        | .mov64 destination source =>
230            .next {advance with addrs := update s.addrs destination (source.eval s)} event
231        | .add64 destination left right =>
232            .next {advance with addrs := update s.addrs destination (left.eval s + right.eval s)} event
233        | .cvt64 destination source =>
234            .next {advance with addrs := update s.addrs destination (BitVec.ofNat 64 (source.eval s).toNat)} event
235        | .setp comparison destination left right =>
236            .next {advance with preds := update s.preds destination (comparison.eval (left.eval s) (right.eval s))} event
237        | .load destination address =>
238            let pointer := address.eval s
239            match addressIndex s.memory pointer with
240            | .error reason => .fault reason
241            | .ok index =>
242                let value := readOverride.getD (s.memory[index]!)
243                .next {advance with regs := update s.regs destination value}
244                  (occurrence s instruction true (some ⟨.load, pointer, value⟩))
245        | .store address source =>
246            let pointer := address.eval s
247            match addressIndex s.memory pointer with
248            | .error reason => .fault reason
249            | .ok index =>
250                let value := source.eval s
251                .next {advance with memory := s.memory.set index value}
252                  (occurrence s instruction true (some ⟨.store, pointer, value⟩))
253        | .bra target => .next {s with pc := target} event
254        | .exit => .halted s event
255        | .unsupported spelling => .unsupported spelling
256      else .next {s with pc := s.pc + 1} (occurrence s instruction false)

261  | some instruction => eval readOverride instruction s

277/-- Fuel counts dispatches, including predicate-false instructions and explicit exit.
278A branch to an invalid PC faults at the next dispatch. Exhaustion is a separate status. -/
279def runWith : Nat → (Nat → Option Word) → List Instr → State → RunResult
280  | 0, _, _, s => ⟨s, .exhausted, []⟩
281  | fuel + 1, oracle, program, s =>
282      match stepWith (oracle 0) program s with
283      | .next next event =>
284          let rest := runWith fuel (fun index => oracle (index + 1)) program next
285          {rest with trace := event :: rest.trace}
286      | .halted final event => ⟨final, .halted, [event]⟩
287      | .fault reason => ⟨s, .fault reason, []⟩
288      | .unsupported spelling => ⟨s, .unsupported spelling, []⟩
289
290def run (fuel : Nat) (program : List Instr) (s : State) : RunResult :=
291  runWith fuel (fun _ => none) program s

293/-- The derivation retains the same oracle positions as the executable runner.
294Exhaustion is explicit; only `.halted` establishes completion. -/
295inductive Runs (program : List Instr) :
296    (Nat → Option Word) → State → RunResult → Prop where
297  | exhausted : Runs program oracle s ⟨s, .exhausted, []⟩
298  | next : stepWith (oracle 0) program s = .next next event →
299      Runs program (fun index => oracle (index + 1)) next rest →
300      Runs program oracle s {rest with trace := event :: rest.trace}
301  | halted : stepWith (oracle 0) program s = .halted final event →
302      Runs program oracle s ⟨final, .halted, [event]⟩
303  | fault : stepWith (oracle 0) program s = .fault reason →
304      Runs program oracle s ⟨s, .fault reason, []⟩
305  | unsupported : stepWith (oracle 0) program s = .unsupported spelling →
306      Runs program oracle s ⟨s, .unsupported spelling, []⟩

337def resume (fuel : Nat) (program : List Instr) (prior : RunResult) : RunResult :=

339    let suffix := run fuel program prior.state

397/-- Splitting a budget preserves exact states, outcomes, and complete dynamic traces. -/
398theorem run_add (first second : Nat) (program : List Instr) (s : State) :
399    run (first + second) program s = resume second program (run first program s) := by
400  induction first generalizing s with
401  | zero => simp [run_zero, resume]
402  | succ first ih =>
403      rw [Nat.succ_add, run_succ, run_succ]
404      cases hs : step program s with
405      | next next event =>
406          simp only
407          rw [ih]
408          simp [resume]
409          split <;> rfl
410      | halted final event => simp [resume]
411      | fault reason => simp [resume]
412      | unsupported spelling => simp [resume]

414/-- Every successful memory effect has passed alignment and arena-bounds checks,
415including effects from candidate loads whose values were overridden. -/
416theorem eval_memory_safe
417    (hstep : eval override instruction s = .next next event)
418    (hmemory : event.memory = some effect) : ValidAddress s.memory effect.address := by
419  cases instruction with
420  | mk guard op =>
421    cases op <;> cases hguard : guard.eval s <;>
422      simp only [eval, hguard, ↓reduceIte] at hstep
423    all_goals first
424      | (obtain ⟨_, rfl⟩ := StepResult.next.inj hstep; simp [occurrence] at hmemory)
425      | contradiction
426      | skip
427    all_goals
428      split at hstep
429      · contradiction
430      · obtain ⟨_, rfl⟩ := StepResult.next.inj hstep
431        simp only [occurrence, Option.some.injEq] at hmemory
432        subst effect
433        exact (addressIndex_ok_iff.mp (by assumption)).1

508/-- Every emitted load/store is safe relative to the initial arena extent, even
509if the run later faults, is unsupported, exhausts its budget, or uses candidate reads. -/
510theorem runWith_trace_safe (fuel : Nat) (oracle : Nat → Option Word)
511    (program : List Instr) (s : State) :
512    ∀ event ∈ (runWith fuel oracle program s).trace,
513      ∀ effect, event.memory = some effect → ValidAddress s.memory effect.address := by
514  induction fuel generalizing oracle s with
515  | zero => simp [runWith]
516  | succ fuel ih =>
517      cases hs : stepWith (oracle 0) program s with
518      | next next emitted =>
519          simp only [runWith, hs, List.mem_cons]
520          intro event member effect memory
521          rcases member with rfl | member
522          · exact step_memory_safe hs memory
523          · have safe := ih _ next event member effect memory
524            simpa only [ValidAddress, step_memory_length hs] using safe
525      | halted final emitted =>
526          simp only [runWith, hs, List.mem_singleton]
527          intro event member effect memory
528          subst event
529          rw [(step_halted hs).2] at memory
530          contradiction
531      | fault reason => simp [runWith, hs]
532      | unsupported spelling => simp [runWith, hs]

543  induction h with
