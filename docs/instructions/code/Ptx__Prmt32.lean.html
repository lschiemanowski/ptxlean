28def compute (_op : Operation) (a b c : Word) : Word :=
29  (BitVec.ofBoolListLE (List.ofFn fun i : Fin 32 =>
30    let selector := (c.toNat / 2 ^ (4 * (i.val / 8))) % 16
31    let source := if selector % 8 < 4 then a else b
32    source.getLsbD (8 * (selector % 4) + (if 8 ≤ selector then 7 else i.val % 8)))).cast (by simp)

34def family : Pure32.Family Operation :=
35  Pure32.Family.ofFunction (fun _ => 3) (fun _ => 0)
36    (fun op words _ => compute op (words w0) (words w1) (words w2))
37    (fun target _ => SupportedTarget target)

39def lower (i : Instr) : Pure32.Instr family where
40  guard := i.guard
41  operation := i.operation
42  destination := i.destination
43  words := fun j => if j.val = 0 then i.first else if j.val = 1 then i.second else i.control
44  predicates := Fin.elim0

46def result (i : Instr) (s : Scalar.State) : Word :=
47  compute i.operation (i.first.eval s) (i.second.eval s) (i.control.eval s)

56theorem compute_bit (op : Operation) (a b c : Word) (bit : Nat) (bound : bit < 32) :
57    (compute op a b c).getLsbD bit = (let selector := (c.toNat / 2 ^ (4 * (bit / 8))) % 16
58     let source := if selector % 8 < 4 then a else b
59     source.getLsbD (8 * (selector % 4) + (if 8 ≤ selector then 7 else bit % 8))) := by
60  cases op
61  simp only [compute, BitVec.getLsbD_cast, BitVec.getLsbD_ofBoolListLE,
62    List.getD_eq_getElem?_getD, List.getElem?_ofFn]
63  simp [bound]

76theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence) (enabled : i.guard.eval s = true) :
77    Eval i s next event ↔ next = Pure32.write s i.destination (result i s) ∧ event = occurrence s i true := by
78  change Pure32.Eval (lower i) s next event ↔ _
79  rw [Pure32.eval_true_iff (lower i) s next event enabled]
80  constructor
81  · rintro ⟨value, h, hn, he⟩
82    have hv : value = result i s := by
83      simpa [family, Pure32.Family.ofFunction, Pure32.Instr.wordValues, Pure32.Instr.predicateValues, lower, result, w0, w1, w2] using h
84    subst value
85    exact ⟨hn, he⟩
86  · rintro ⟨hn, he⟩
87    refine ⟨result i s, ?_, hn, he⟩
88    simp [family, Pure32.Family.ofFunction, Pure32.Instr.wordValues, lower, result, w0, w1, w2]

114theorem step_origin (target : Ptx.Target) (program : List Instr) (s next : Scalar.State) (event : Occurrence) (stepped : Step target program s next event) :
115    SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧ event.pc = s.pc ∧ event.instruction = lower i ∧ Eval i s next event := by
116  obtain ⟨j, fetch, supported, pc, ins, evaluated⟩ := Pure32.step_origin stepped
117  have original : ∃ i, program[s.pc]? = some i ∧ lower i = j := by
118    cases hx : program[s.pc]? with
119    | none => simp [hx] at fetch
120    | some i => have hm : some (lower i) = some j := by simpa [hx] using fetch
121                exact ⟨i, by simp, Option.some.inj hm⟩
122  obtain ⟨i, fi, eq⟩ := original
123  have ev : Eval i s next event := Eq.mp (congrArg (fun x : Pure32.Instr family => Pure32.Eval x s next event) eq.symm) evaluated
124  exact ⟨supported, i, fi, pc, ins.trans eq.symm, ev⟩

142def encode (i : Instr) : Scalar.Text.Statement :=
143  ⟨i.guard, "prmt.b32", [.word (.reg i.destination), .word i.first, .word i.second, .word i.control]⟩

145def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
146  if supportedMnemonic statement.mnemonic then
147    match statement.operands with
148    | [.word (.reg d), .word a, .word b, .word c] => .ok ⟨statement.guard, .permute, d, a, b, c⟩
149    | _ => .error (.invalidOperands statement.mnemonic)
150  else .error (.unsupportedMnemonic statement.mnemonic)

152theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
153  cases i with | mk guard operation destination first second control => cases operation <;> rfl

155theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) : decode statement = .ok i ↔ statement = encode i := by
156  constructor
157  · intro h
158    rcases statement with ⟨guard, mnemonic, operands⟩
159    cases i with
160    | mk g op d a b c =>
161      cases op
162      unfold decode at h
163      by_cases hm : mnemonic = "prmt.b32"
164      · subst mnemonic
165        simp [supportedMnemonic] at h
166        split at h
167        · injection h with hi
168          cases hi
169          rfl
170        · simp at h
171      · simp [supportedMnemonic, hm] at h
172  · intro h
173    subst statement
174    exact decode_encode i
