30def compute (_op : Operation) (a b c d : Word) : Word :=
31  (BitVec.ofBoolListLE (List.ofFn fun (i : Fin 32) =>
32    if c.toNat % 256 ≤ i.val ∧ i.val < c.toNat % 256 + d.toNat % 256 then
33      a.getLsbD (i.val - c.toNat % 256) else b.getLsbD i.val)).cast (by simp)

35def family : Pure32.Family Operation :=
36  Pure32.Family.ofFunction (fun _ => 4) (fun _ => 0)
37    (fun op words _ => compute op (words w0) (words w1) (words w2) (words w3))
38    (fun target _ => SupportedTarget target)

40def lower (i : Instr) : Pure32.Instr family where
41  guard := i.guard
42  operation := i.operation
43  destination := i.destination
44  words := fun j => if j.val = 0 then i.insert else if j.val = 1 then i.base else if j.val = 2 then i.position else i.length
45  predicates := Fin.elim0

59theorem compute_bit (op : Operation) (a b c d : Word) (bit : Nat) (bound : bit < 32) :
60    (compute op a b c d).getLsbD bit = (if c.toNat % 256 ≤ bit ∧ bit < c.toNat % 256 + d.toNat % 256 then a.getLsbD (bit - c.toNat % 256) else b.getLsbD bit) := by
61  cases op
62  simp only [compute, BitVec.getLsbD_cast, BitVec.getLsbD_ofBoolListLE,
63    List.getD_eq_getElem?_getD, List.getElem?_ofFn]
64  simp [bound]

77theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence) (enabled : i.guard.eval s = true) :
78    Eval i s next event ↔ next = Pure32.write s i.destination (result i s) ∧ event = occurrence s i true := by
79  change Pure32.Eval (lower i) s next event ↔ _
80  rw [Pure32.eval_true_iff (lower i) s next event enabled]
81  constructor
82  · rintro ⟨value, h, hn, he⟩
83    have hv : value = result i s := by
84      simpa [family, Pure32.Family.ofFunction, Pure32.Instr.wordValues, Pure32.Instr.predicateValues,
85        lower, fin3_val, result, w0, w1, w2, w3] using h
86    subst value
87    exact ⟨hn, he⟩
88  · rintro ⟨hn, he⟩
89    refine ⟨result i s, ?_, hn, he⟩
90    simp [family, Pure32.Family.ofFunction, Pure32.Instr.wordValues, lower, fin3_val, result, w0, w1, w2, w3]

116theorem step_origin (target : Ptx.Target) (program : List Instr) (s next : Scalar.State) (event : Occurrence) (stepped : Step target program s next event) :
117    SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧ event.pc = s.pc ∧ event.instruction = lower i ∧ Eval i s next event := by
118  obtain ⟨j, fetch, supported, pc, ins, evaluated⟩ := Pure32.step_origin stepped
119  have original : ∃ i, program[s.pc]? = some i ∧ lower i = j := by
120    cases hx : program[s.pc]? with
121    | none => simp [hx] at fetch
122    | some i => have hm : some (lower i) = some j := by simpa [hx] using fetch
123                exact ⟨i, by simp, Option.some.inj hm⟩
124  obtain ⟨i, fi, eq⟩ := original
125  have ev : Eval i s next event := Eq.mp (congrArg (fun x : Pure32.Instr family => Pure32.Eval x s next event) eq.symm) evaluated
126  exact ⟨supported, i, fi, pc, ins.trans eq.symm, ev⟩

147def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
148  if supportedMnemonic statement.mnemonic then
149    match statement.operands with
150    | [.word (.reg destination), .word insert, .word base, .word position, .word length] => .ok ⟨statement.guard, .insert, destination, insert, base, position, length⟩
151    | _ => .error (.invalidOperands statement.mnemonic)
152  else .error (.unsupportedMnemonic statement.mnemonic)

157theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) : decode statement = .ok i ↔ statement = encode i := by
158  constructor
159  · intro h
160    rcases statement with ⟨guard, mnemonic, operands⟩
161    simp only [decode] at h
162    by_cases hm : mnemonic = "bfi.b32"
163    · subst mnemonic
164      simp [supportedMnemonic] at h
165      split at h
166      · injection h with hi
167        cases hi
168        rfl
169      · simp at h
170    · simp [supportedMnemonic, hm] at h
