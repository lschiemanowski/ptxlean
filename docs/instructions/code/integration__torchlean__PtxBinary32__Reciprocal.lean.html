12def reference (input : Word) : Word :=
13  (Model.div (Model.posOne FloatFormat.binary32) (Model.ofBits (fmt := FloatFormat.binary32) input)).bits

15def Results (input output : Word) : Prop := Envelope (reference input) output

17instance instDecidableResults (input output : Word) : Decidable (Results input output) := by

19  infer_instance

21theorem results_exists (input : Word) : ∃ output, Results input output := envelope_exists _

23theorem input_nonzero (input : Word) (x : ℝ) (value : finiteReal input = some x)
24    (nonzero : x ≠ 0) : Model.isZero (Model.ofBits (fmt := FloatFormat.binary32) input) = false := by
25  cases h : Model.isZero (Model.ofBits (fmt := FloatFormat.binary32) input)
26  · rfl
27  · have zero := Model.toReal_eq_zero_of_isZero _ h
28    have val : Model.toReal (Model.ofBits (fmt := FloatFormat.binary32) input) = x :=
29      (Bounds.finiteReal_spec input x value).2
30    exact False.elim (nonzero (val.symm.trans zero))

32theorem reference_finite (input : Word) (x : ℝ) (value : finiteReal input = some x)
33    (nonzero : x ≠ 0) (range : |1/x| ≤ Bounds.maxFinite) :
34    isFinite (reference input) = true := by
35  have spec : Model.isFinite (Model.ofBits (fmt := FloatFormat.binary32) input) = true ∧
36      Model.toReal (Model.ofBits (fmt := FloatFormat.binary32) input) = x :=
37    Bounds.finiteReal_spec input x value
38  have h := Model.isFinite_div_of_abs_div_le_posMaxFinite
39    (Model.posOne FloatFormat.binary32) (Model.ofBits (fmt := FloatFormat.binary32) input) (by decide)
40    (by decide) spec.1 (input_nonzero input x value nonzero)
41    (by rw [Model.toReal_posOne, spec.2]; exact range)
42  change Model.isFinite (Binary.toModel (decode (reference input))) = true
43  rw [reference, RoundedReal.decode_bits]
44  exact h

46theorem reference_real (input : Word) (x : ℝ) (value : finiteReal input = some x)
47    (nonzero : x ≠ 0) (range : |1/x| ≤ Bounds.maxFinite) :
48    finiteReal (reference input) = some (round32 (1/x)) := by
49  rw [finiteReal_some _ (reference_finite input x value nonzero range)]
50  rw [reference, RoundedReal.decode_bits]
51  apply congrArg some
52  have spec : Model.isFinite (Model.ofBits (fmt := FloatFormat.binary32) input) = true ∧
53      Model.toReal (Model.ofBits (fmt := FloatFormat.binary32) input) = x :=
54    Bounds.finiteReal_spec input x value
55  have h := Model.toReal_div_eq_roundAt_of_abs_div_le_posMaxFinite
56    (Model.posOne FloatFormat.binary32) (Model.ofBits (fmt := FloatFormat.binary32) input) (by decide)
57    (by decide) spec.1 (input_nonzero input x value nonzero)
58    (by rw [Model.toReal_posOne, spec.2]; exact range)
59  rw [Model.toReal_posOne, spec.2] at h
60  exact h

62theorem results_real (input output : Word) (x : ℝ) (value : finiteReal input = some x)
63    (nonzero : x ≠ 0) (range : |1/x| ≤ Bounds.maxFinite) (allowed : Results input output) :
64    finiteReal output = some (round32 (1/x)) := by
65  rw [envelope_finite_eq _ _ (reference_finite input x value nonzero range) allowed]
66  exact reference_real input x value nonzero range

68theorem results_error (input output : Word) (x : ℝ) (value : finiteReal input = some x)
69    (nonzero : x ≠ 0) (range : |1/x| ≤ Bounds.maxFinite) (allowed : Results input output) :
70    ∃ z : ℝ, finiteReal output = some z ∧ |z-1/x| ≤ eps32 (1/x) :=
71  ⟨_, results_real input output x value nonzero range allowed,
72    TorchLean.Floats.FP32.round_abs_error _⟩

74/-- Sufficient for a softmax denominator whose total error is at most one half. -/
75theorem half_range (x : ℝ) (lower : 1/2 ≤ x) : |1/x| ≤ Bounds.maxFinite := by
76  have pos : 0 < x := by linarith
77  have upper : 1/x ≤ 2 := (div_le_iff₀ pos).mpr (by linarith)
78  rw [abs_of_nonneg (div_nonneg (by norm_num) pos.le)]
79  exact upper.trans (by rw [Bounds.maxFinite_eq]; norm_num)
80
81theorem finite_result_exists (input : Word) (x : ℝ) (value : finiteReal input = some x)
82    (lower : 1/2 ≤ x) :
83    ∃ output, Results input output ∧ finiteReal output = some (round32 (1/x)) :=
84  ⟨reference input, envelope_self _, reference_real input x value (by linarith) (half_range x lower)⟩

86/-- Derive the reciprocal's range from the perturbed denominator, without
87assuming a successful reciprocal or its output finiteness. -/
88theorem result_exists_of_sum_error (input : Word) (S Z D : ℝ)
89    (value : finiteReal input = some S) (exactLower : 1 ≤ Z)
90    (error : |S-Z| ≤ D) (small : D ≤ 1/2) :
91    ∃ output, Results input output ∧ finiteReal output = some (round32 (1/S)) := by
92  apply finite_result_exists input S value
93  have := abs_le.mp error
94  linarith
