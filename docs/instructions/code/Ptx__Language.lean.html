3/-!
4# Local execution for the message-passing fragment
5
6This typed, straight-line language has aligned, non-overlapping 32-bit accesses
7to a single global-memory arena. Addresses are word indices, not raw pointers.
8Stores use immediate operands. Loads receive a candidate value from an oracle;
9the global memory model separately decides whether those values are permitted.
10There is no branch, address arithmetic, parsing, allocation, or GPU scheduler.
11-/

34structure Effect where

37  value : Word

40inductive Instr where
41  | load (order : LoadOrder) (address destination : Nat)
42  | store (order : StoreOrder) (address : Nat) (value : Word)
43  deriving DecidableEq, Repr

45def Instr.address : Instr → Nat
46  | .load _ address _ => address
47  | .store _ address _ => address
48
49/-- The memory event requested by an instruction, before global validation. -/
50def Instr.effect (instruction : Instr) (readValue : Word) : Effect :=
51  match instruction with
52  | .load order address _ => ⟨.load order, address, readValue⟩
53  | .store order address value => ⟨.store order, address, value⟩
54
55/-- One local instruction. Only loads consume the candidate read value. -/
56def Instr.step (instruction : Instr) (registers : Registers) (readValue : Word) :
57    Registers × Effect :=
58  match instruction with
59  | .load order address destination =>
60      (fun register => if register = destination then readValue else registers register,
61       ⟨.load order, address, readValue⟩)
62  | .store order address value =>
63      (registers, ⟨.store order, address, value⟩)
64
65/-- A local transition is precisely the graph of the instruction operation. -/
66def Step (instruction : Instr) (registers : Registers) (readValue : Word)
67    (next : Registers) (effect : Effect) : Prop :=
68  instruction.step registers readValue = (next, effect)
69
70theorem step_exists (instruction : Instr) (registers : Registers) (readValue : Word) :
71    ∃ next effect, Step instruction registers readValue next effect := by
72  exact ⟨(instruction.step registers readValue).1,
73    (instruction.step registers readValue).2, rfl⟩
74
75theorem step_deterministic
76    (h₁ : Step instruction registers readValue next₁ effect₁)
77    (h₂ : Step instruction registers readValue next₂ effect₂) :
78    next₁ = next₂ ∧ effect₁ = effect₂ := by
79  have h : (next₁, effect₁) = (next₂, effect₂) := h₁.symm.trans h₂
80  exact Prod.mk.inj h

112/-- Oracle position is instruction position; values at store positions are ignored. -/
113def execute : List Instr → Registers → (Nat → Word) → Registers × List Effect
114  | [], registers, _ => (registers, [])
115  | instruction :: rest, registers, oracle =>
116      let (next, effect) := instruction.step registers (oracle 0)
117      let (final, effects) := execute rest next (fun i => oracle (i + 1))
118      (final, effect :: effects)
119
120/-- Finite local executions compose instruction transitions in program order. -/
121inductive Runs : List Instr → Registers → (Nat → Word) → Registers → List Effect → Prop
122  | nil (registers : Registers) (oracle : Nat → Word) :
123      Runs [] registers oracle registers []
124  | cons : Step instruction registers (oracle 0) next effect →
125      Runs rest next (fun i => oracle (i + 1)) final effects →
126      Runs (instruction :: rest) registers oracle final (effect :: effects)
127
128theorem execute_runs (program : List Instr) (registers : Registers) (oracle : Nat → Word) :
129    Runs program registers oracle (execute program registers oracle).1
130      (execute program registers oracle).2 := by
131  induction program generalizing registers oracle with
132  | nil => exact .nil registers oracle
133  | cons instruction rest ih =>
134      exact .cons rfl (ih _ _)
135
136theorem runs_eq_execute
137    (h : Runs program registers oracle final effects) :
138    execute program registers oracle = (final, effects) := by
139  induction h with
140  | nil => rfl
141  | @cons instruction registers oracle next effect rest final effects hs hr ih =>
142      simp only [execute, Step] at *
143      rw [hs]
144      simp only [ih]
145
146theorem runs_exists (program : List Instr) (registers : Registers) (oracle : Nat → Word) :
147    ∃ final effects, Runs program registers oracle final effects :=
148  ⟨_, _, execute_runs program registers oracle⟩
149
150theorem runs_deterministic
151    (h₁ : Runs program registers oracle final₁ effects₁)
152    (h₂ : Runs program registers oracle final₂ effects₂) :
153    final₁ = final₂ ∧ effects₁ = effects₂ := by
154  exact Prod.mk.inj ((runs_eq_execute h₁).symm.trans (runs_eq_execute h₂))

156theorem execute_addresses (program : List Instr) (registers : Registers) (oracle : Nat → Word) :
157    ((execute program registers oracle).2.map Effect.address) = program.map Instr.address := by
158  induction program generalizing registers oracle with
159  | nil => rfl
160  | cons instruction rest ih =>
161      simp only [execute, List.map_cons, step_address, ih]

163/-- Every instruction emits exactly one event, in instruction order. -/
164theorem execute_effects (program : List Instr) (registers : Registers) (oracle : Nat → Word) :
165    (execute program registers oracle).2 =
166      program.mapIdx (fun i instruction => instruction.effect (oracle i)) := by
167  induction program generalizing registers oracle with
168  | nil => simp [execute]
169  | cons instruction rest ih =>
170      simp only [execute, step_effect, List.mapIdx_cons, ih]

174    (execute program registers oracle).2.length = program.length := by

178theorem execute_getElem? (program : List Instr) (registers : Registers)

185/-- Byte addresses are mathematical naturals; no finite-pointer arithmetic is modeled. -/
186def byteAddress (wordIndex : Nat) : Nat := 4 * wordIndex
187
188def AccessSafe (nWords : Nat) (effect : Effect) : Prop :=
189  byteAddress effect.address % 4 = 0 ∧
190    byteAddress effect.address + 4 ≤ 4 * nWords
191
192theorem access_safe_of_index_lt (effect : Effect) (h : effect.address < nWords) :
193    AccessSafe nWords effect := by
194  constructor
195  · simp [byteAddress]
196  · simp only [byteAddress]
197    omega
198
199/-- Arena bounds on typed instructions imply alignment and bounds for all emitted accesses. -/
200theorem execute_access_safe (program : List Instr) (registers : Registers) (oracle : Nat → Word)
201    (bounded : ∀ instruction ∈ program, instruction.address < nWords) :
202    ∀ effect ∈ (execute program registers oracle).2, AccessSafe nWords effect := by
203  intro effect he
204  apply access_safe_of_index_lt
205  have ha : effect.address ∈ (execute program registers oracle).2.map Effect.address :=
206    List.mem_map.mpr ⟨effect, he, rfl⟩
207  rw [execute_addresses] at ha
208  obtain ⟨instruction, hi, haddr⟩ := List.mem_map.mp ha
209  rw [← haddr]
210  exact bounded instruction hi

212def InWordFootprint (wordIndex byte : Nat) : Prop :=
213  ∃ offset, offset < 4 ∧ byte = byteAddress wordIndex + offset
214
215/-- Distinct word indices name disjoint complete four-byte footprints. -/
216theorem word_footprints_disjoint (different : left ≠ right)
217    (hl : InWordFootprint left byte) (hr : InWordFootprint right byte) : False := by
218  obtain ⟨lo, hlo, heql⟩ := hl
219  obtain ⟨ro, hro, heqr⟩ := hr
220  simp only [byteAddress] at *
221  omega
