8structure Instr where

10  destination : Nat
11  source : Scalar.Operand32

14structure Occurrence where
15  pc : Nat
16  instruction : Instr
17  executed : Bool
18  reads : List Scalar.Register
19  writes : List Scalar.Register
20  memory : Option Scalar.MemoryEffect
21  deriving DecidableEq, Repr

23def occurrence (s : Scalar.State) (i : Instr) (executed : Bool) : Occurrence :=
24  ⟨s.pc, i, executed, i.guard.reads ++ (if executed then i.source.reads else []),
25    (if executed then [.word i.destination] else []), none⟩
26
27def SupportedTarget (target : Ptx.Target) : Prop :=
28  target.isa = 94 ∧ 20 ≤ target.sm
29
30inductive Eval (i : Instr) : Scalar.State → Scalar.State → Occurrence → Prop
31  | skipped {s} (disabled : i.guard.eval s = false) :
32      Eval i s {s with pc := s.pc + 1} (occurrence s i false)
33  | executed {s value} (enabled : i.guard.eval s = true)
34      (result : Ptx.Binary32.Reciprocal.Results (i.source.eval s) value) :
35      Eval i s {{s with pc := s.pc + 1} with
36        regs := Scalar.update s.regs i.destination value} (occurrence s i true)
37
38def Step (target : Ptx.Target) (program : List Instr)
39    (s next : Scalar.State) (event : Occurrence) : Prop :=
40  SupportedTarget target ∧ ∃ i, program[s.pc]? = some i ∧ Eval i s next event

42theorem eval_true_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
43    (enabled : i.guard.eval s = true) :
44    Eval i s next event ↔ ∃ value,
45      Ptx.Binary32.Reciprocal.Results (i.source.eval s) value ∧
46      next = {s with pc := s.pc+1, regs := Scalar.update s.regs i.destination value} ∧
47      event = occurrence s i true := by
48  constructor
49  · intro h; cases h with
50    | skipped disabled => simp [enabled] at disabled
51    | executed _ result => exact ⟨_, result, rfl, rfl⟩
52  · rintro ⟨value, result, rfl, rfl⟩; exact .executed enabled result
53
54theorem eval_false_iff (i : Instr) (s next : Scalar.State) (event : Occurrence)
55    (disabled : i.guard.eval s = false) :
56    Eval i s next event ↔ next = {s with pc := s.pc+1} ∧ event = occurrence s i false := by
57  constructor
58  · intro h; cases h with
59    | skipped _ => exact ⟨rfl, rfl⟩
60    | executed enabled _ => simp [disabled] at enabled
61  · rintro ⟨rfl, rfl⟩; exact .skipped disabled

63theorem eval_destination (i : Instr) (s next : Scalar.State) (event : Occurrence)
64    (hEval : Eval i s next event) (enabled : i.guard.eval s = true) :
65    Ptx.Binary32.Reciprocal.Results (i.source.eval s) (next.regs i.destination) := by
66  cases hEval with
67  | skipped disabled => simp [enabled] at disabled
68  | executed _ result => simpa using result

70theorem eval_frame (i : Instr) (s next : Scalar.State) (event : Occurrence)
71    (hEval : Eval i s next event) :
72    next.pc = s.pc+1 ∧ next.memory = s.memory ∧ next.addrs = s.addrs ∧ next.preds = s.preds := by
73  cases hEval <;> simp_all

75theorem eval_other (i : Instr) (s next : Scalar.State) (event : Occurrence) (other : Nat)
76    (hEval : Eval i s next event) (different : other ≠ i.destination) :
77    next.regs other = s.regs other := by
78  cases hEval with
79  | skipped _ => rfl
80  | executed _ _ => simp [Scalar.update, different]

82theorem eval_event (i : Instr) (s next : Scalar.State) (event : Occurrence)
83    (hEval : Eval i s next event) : event = occurrence s i (i.guard.eval s) := by
84  cases hEval with
85  | skipped h => simp [h]
86  | executed h _ => simp [h]

88theorem eval_exists (i : Instr) (s : Scalar.State) : ∃ next event, Eval i s next event := by
89  by_cases h : i.guard.eval s = true
90  · obtain ⟨value, hv⟩ := Ptx.Binary32.Reciprocal.results_exists (i.source.eval s)
91    exact ⟨{s with pc := s.pc+1, regs := Scalar.update s.regs i.destination value},
92      occurrence s i true, .executed h hv⟩
93  · have hf : i.guard.eval s = false := Bool.eq_false_of_not_eq_true h
94    exact ⟨{s with pc := s.pc+1}, occurrence s i false, .skipped hf⟩

96theorem step_origin (target : Ptx.Target) (program : List Instr)
97    (s next : Scalar.State) (event : Occurrence) (hStep : Step target program s next event) :
98    SupportedTarget target ∧ ∃ i,
99      program[s.pc]? = some i ∧ event.instruction = i ∧ Eval i s next event := by
100  rcases hStep with ⟨hs, i, fetch, he⟩
101  exact ⟨hs, i, fetch, by cases he <;> rfl, he⟩
102
103theorem step_exists (target : Ptx.Target) (program : List Instr) (s : Scalar.State)
104    (i : Instr) (supported : SupportedTarget target) (fetch : program[s.pc]? = some i) :
105    ∃ next event, Step target program s next event := by
106  obtain ⟨next, event, he⟩ := eval_exists i s
107  exact ⟨next, event, supported, i, fetch, he⟩

109theorem step_no_fetch (target : Ptx.Target) (program : List Instr)
110    (s next : Scalar.State) (event : Occurrence) (missing : program[s.pc]? = none) :
111    ¬ Step target program s next event := by
112  rintro ⟨_, i, fetch, _⟩; rw [missing] at fetch; contradiction

114theorem step_unsupported_target (target : Ptx.Target) (program : List Instr)

116    ¬ Step target program s next event := by

121def mnemonic : String := "rcp.rn.f32"
122def supportedMnemonic (name : String) : Bool := name == "rcp.rn.f32"
123
124def encode (i : Instr) : Scalar.Text.Statement :=
125  ⟨i.guard, mnemonic, [.word (.reg i.destination), .word i.source]⟩
126
127def decode (statement : Scalar.Text.Statement) : Except Scalar.Text.DecodeError Instr :=
128  if statement.mnemonic == mnemonic then
129    match statement.operands with
130    | [.word (.reg d), .word source] => .ok ⟨statement.guard, d, source⟩
131    | _ => .error (.invalidOperands statement.mnemonic)
132  else .error (.unsupportedMnemonic statement.mnemonic)
133
134theorem decode_encode (i : Instr) : decode (encode i) = .ok i := by
135  cases i with | mk g d s => rfl
136
137theorem decode_iff (statement : Scalar.Text.Statement) (i : Instr) :
138    decode statement = .ok i ↔ statement = encode i := by
139  constructor
140  · intro h
141    cases statement with
142    | mk g m ops =>
143      cases i with
144      | mk gi di si =>
145        by_cases hm : m = mnemonic
146        · subst m
147          simp only [decode] at h
148          cases ops with
149          | nil => simp [decode] at h
150          | cons first rest => cases first with
151            | word first => cases first with
152              | reg d => cases rest with
153                | nil => simp [decode] at h
154                | cons second rest2 => cases second with
155                  | word source => cases rest2 with
156                    | nil =>
157                        simp only [decode] at h
158                        rcases h with ⟨rfl, rfl, rfl⟩
159                        rfl
160                    | cons _ _ => simp [decode] at h
161                  | _ => simp [decode] at h
162              | imm _ => simp [decode] at h
163            | _ => simp [decode] at h
164        · simp [decode, hm] at h
165  · rintro rfl; exact decode_encode i
