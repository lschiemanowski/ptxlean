5/-!
6# Typed mnemonic boundary
7
8This decoder consumes a mnemonic and already-tokenized, typed operands. It is not
9a raw PTX lexer/parser, module parser, register-declaration checker, or label
10resolver. `label` carries a resolved instruction index. Canonical encoding has a
11checked round trip for every supported instruction, including its predicate.
12-/
13namespace Ptx.Scalar.Text

15inductive Token where
16  | word (operand : Operand32)
17  | address (operand : Operand64)
18  | predicate (index : Nat)
19  | memory (address : Operand64)
20  | label (resolvedPC : Nat)
21  deriving DecidableEq, Repr
22
23structure Statement where
24  guard : Guard := .always
25  mnemonic : String
26  operands : List Token
27  deriving DecidableEq, Repr

29inductive DecodeError where
30  | unsupportedMnemonic (mnemonic : String)
31  | invalidOperands (mnemonic : String)
32  deriving DecidableEq, Repr

34def binMnemonic : BinOp → String

36  | .sub => "sub.u32"
37  | .mulLo => "mul.lo.u32"
38  | .and => "and.b32"

40  | .xor => "xor.b32"
41  | .shl => "shl.b32"
42  | .shr => "shr.u32"
43  | .minU => "min.u32"
44  | .maxU => "max.u32"

58def supportedMnemonic (mnemonic : String) : Bool :=
59  mnemonic ∈ ["mov.b32", "add.u32", "sub.u32", "mul.lo.u32", "and.b32", "or.b32", "xor.b32",
60    "shl.b32", "shr.u32", "min.u32", "max.u32", "clz.b32", "popc.b32", "mov.b64", "add.u64", "cvt.u64.u32", "setp.eq.u32", "setp.ne.u32",
61    "setp.lt.u32", "setp.le.u32", "setp.gt.u32", "setp.ge.u32",
62    "ld.relaxed.gpu.global.u32", "st.relaxed.gpu.global.u32", "bra", "exit"]

64def decodeOp (mnemonic : String) (operands : List Token) : Except DecodeError Op :=
65  match mnemonic, operands with
66  | "mov.b32", [.word (.reg d), .word s] => .ok (.mov32 d s)
67  | "add.u32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .add d a b)
68  | "sub.u32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .sub d a b)
69  | "mul.lo.u32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .mulLo d a b)
70  | "and.b32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .and d a b)
71  | "or.b32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .or d a b)
72  | "xor.b32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .xor d a b)
73  | "shl.b32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .shl d a b)
74  | "shr.u32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .shr d a b)
75  | "min.u32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .minU d a b)
76  | "max.u32", [.word (.reg d), .word a, .word b] => .ok (.bin32 .maxU d a b)
77  | "clz.b32", [.word (.reg d), .word a] => .ok (.unary32 .clz d a)
78  | "popc.b32", [.word (.reg d), .word a] => .ok (.unary32 .popc d a)
79  | "mov.b64", [.address (.reg d), .address s] => .ok (.mov64 d s)
80  | "add.u64", [.address (.reg d), .address a, .address b] => .ok (.add64 d a b)
81  | "cvt.u64.u32", [.address (.reg d), .word s] => .ok (.cvt64 d s)
82  | "setp.eq.u32", [.predicate d, .word a, .word b] => .ok (.setp .eq d a b)
83  | "setp.ne.u32", [.predicate d, .word a, .word b] => .ok (.setp .ne d a b)
84  | "setp.lt.u32", [.predicate d, .word a, .word b] => .ok (.setp .lt d a b)
85  | "setp.le.u32", [.predicate d, .word a, .word b] => .ok (.setp .le d a b)
86  | "setp.gt.u32", [.predicate d, .word a, .word b] => .ok (.setp .gt d a b)
87  | "setp.ge.u32", [.predicate d, .word a, .word b] => .ok (.setp .ge d a b)
88  | "ld.relaxed.gpu.global.u32", [.word (.reg d), .memory address] => .ok (.load d address)
89  | "st.relaxed.gpu.global.u32", [.memory address, .word (.reg source)] => .ok (.store address (.reg source))
90  | "bra", [.label target] => .ok (.bra target)
91  | "exit", [] => .ok .exit
92  | mnemonic, _ => if supportedMnemonic mnemonic then .error (.invalidOperands mnemonic)
93      else .error (.unsupportedMnemonic mnemonic)

95def decode (statement : Statement) : Except DecodeError Instr :=
96  (decodeOp statement.mnemonic statement.operands).map fun op => ⟨statement.guard, op⟩
97
98def encodeOp : Op → String × List Token
99  | .mov32 d s => ("mov.b32", [.word (.reg d), .word s])
100  | .bin32 operation d a b => (binMnemonic operation, [.word (.reg d), .word a, .word b])
101  | .unary32 operation d a => (unaryMnemonic operation, [.word (.reg d), .word a])
102  | .mov64 d s => ("mov.b64", [.address (.reg d), .address s])
103  | .add64 d a b => ("add.u64", [.address (.reg d), .address a, .address b])
104  | .cvt64 d s => ("cvt.u64.u32", [.address (.reg d), .word s])
105  | .setp comparison d a b => (compareMnemonic comparison, [.predicate d, .word a, .word b])
106  | .load d address => ("ld.relaxed.gpu.global.u32", [.word (.reg d), .memory address])
107  | .store address (.reg source) => ("st.relaxed.gpu.global.u32", [.memory address, .word (.reg source)])
108  | .store address (.imm value) => ("internal.store-immediate.u32", [.memory address, .word (.imm value)])
109  | .bra target => ("bra", [.label target])
110  | .exit => ("exit", [])
111  | .unsupported spelling => (spelling, [])
112
113def encode (instruction : Instr) : Statement :=
114  ⟨instruction.guard, (encodeOp instruction.op).1, (encodeOp instruction.op).2⟩

116/-- Literal stores are an internal machine convenience, not a legal `st` source. -/
117def SupportedOp : Op → Prop
118  | .unsupported _ | .store _ (.imm _) => False
119  | _ => True
120
121def Supported (instruction : Instr) : Prop := SupportedOp instruction.op
122
123theorem decode_encode (instruction : Instr) (supported : Supported instruction) :
124    decode (encode instruction) = .ok instruction := by
125  rcases instruction with ⟨guard, op⟩
126  cases op with
127  | bin32 operation d a b => cases operation <;> rfl
128  | unary32 operation d a => cases operation <;> rfl
129  | setp comparison d a b => cases comparison <;> rfl
130  | store address source =>
131      cases source with
132      | reg index => rfl
133      | imm value => exact False.elim supported
134  | unsupported spelling => exact False.elim supported
135  | _ => rfl

137/-- Unknown spellings remain explicit errors, including bare weak-memory forms. -/
138theorem bare_load_rejected : decodeOp "ld.global.u32" [] =
139    .error (.unsupportedMnemonic "ld.global.u32") := rfl

149/- Decoding cannot manufacture an unsupported operation or an immediate store. -/
150theorem decodeOp_supported (h : decodeOp mnemonic operands = .ok op) : SupportedOp op := by
151  unfold decodeOp at h
152  split at h
153  all_goals first
154    | (obtain rfl := Except.ok.inj h; trivial)
155    | (split at h <;> contradiction)
156
157/-- PTX `st` requires its data operand in the register state space. -/
158theorem immediate_store_rejected :
159    decodeOp "st.relaxed.gpu.global.u32" [.memory (.reg 0), .word (.imm 7)] =
160      .error (.invalidOperands "st.relaxed.gpu.global.u32") := rfl

162theorem decode_supported (h : decode statement = .ok instruction) : Supported instruction := by
163  unfold decode at h
164  cases decoded : decodeOp statement.mnemonic statement.operands with
165  | error reason => simp [decoded, Except.map] at h
166  | ok op =>
167      simp only [decoded, Except.map, Except.ok.injEq] at h
168      subst instruction
169      exact decodeOp_supported decoded
