3/-! Partial arithmetic provider for the selected non-FTZ approximate exponential.
4Positive finite operands are unsupported, not undefined. The finite allowance
5covers the two named ULP interpretations; it does not resolve source ambiguity.
6There is no fetched instruction or execution semantics in this module. -/

10/-- Supported operands include every special value and finite nonpositive reals. -/
11def Supported (input : Word) : Prop :=
12  isNaN input = true ∨ input = 0xff800000 ∨ input = 0x7f800000 ∨
13    ∃ x : ℝ, finiteReal input = some x ∧ x ≤ 0
14
15/-- Exact special cases take precedence over a finite approximate allowance.
16The output is unrestricted within the conservative envelope for negative inputs. -/
17def Results (input output : Word) : Prop :=
18  (isNaN input = true ∧ isNaN output = true) ∨
19  (input = 0xff800000 ∧ output = 0) ∨
20  (input = 0x7f800000 ∧ output = 0x7f800000) ∨
21  ∃ x : ℝ, finiteReal input = some x ∧ x ≤ 0 ∧
22    if x = 0 then output = 0x3f800000 else ConservativeULP.Results ((2:ℝ)^x) output
23
24private theorem finite_ne_infinities (input : Word) (x : ℝ)
25    (value : finiteReal input = some x) : input ≠ 0xff800000 ∧ input ≠ 0x7f800000 := by
26  constructor <;> intro h <;> subst input <;>
27    rw [finiteReal_none _ (by decide)] at value <;> contradiction
28
29/-- On a finite operand this characterizes every result, including exact zero. -/
30theorem finite_iff (input output : Word) (x : ℝ) (value : finiteReal input = some x) :
31    Results input output ↔ x ≤ 0 ∧
32      if x = 0 then output = 0x3f800000 else ConservativeULP.Results ((2:ℝ)^x) output := by
33  have hn := finite_not_nan input (Bounds.finiteReal_spec input x value).1
34  have hi := finite_ne_infinities input x value
35  simp only [Results, hn, Bool.false_eq_true, false_and, false_or,
36    hi.1, hi.2, value, Option.some.injEq]
37  simp
38
39/-- Zero has the exact positive-one result, regardless of the input zero sign. -/
40theorem zero_iff (input output : Word) (value : finiteReal input = some 0) :
41    Results input output ↔ output = 0x3f800000 := by
42  simp [finite_iff input output 0 value]
43
44/-- Every envelope output remains admitted; there is no hidden nonnegativity or
45normal-number restriction on returned finite values. -/
46theorem negative_iff (input output : Word) (x : ℝ)
47    (value : finiteReal input = some x) (negative : x < 0) :
48    Results input output ↔ ConservativeULP.Results ((2:ℝ)^x) output := by
49  rw [finite_iff input output x value]
50  simp [negative.le, ne_of_lt negative]
51
52/-- Positive finite arguments have no result in this partial provider. -/
53theorem positive_unsupported (input : Word) (x : ℝ)
54    (value : finiteReal input = some x) (positive : 0 < x) : ¬ Supported input := by
55  have hn := finite_not_nan input (Bounds.finiteReal_spec input x value).1
56  have hi := finite_ne_infinities input x value
57  simp only [Supported, hn, hi.1, hi.2, value, Bool.false_eq_true, false_or]
58  simp [not_le.mpr positive]
59
60/-- NaN payload freedom is conservative: individual payload realizability is not asserted. -/
61theorem nan_iff (input output : Word) (nan : isNaN input = true) :
62    Results input output ↔ isNaN output = true := by
63  have ni : input ≠ 0xff800000 := by intro h; subst input; contradiction
64  have pi : input ≠ 0x7f800000 := by intro h; subst input; contradiction
65  have nv : ∀ x : ℝ, finiteReal input ≠ some x := by
66    intro x value
67    have := finite_not_nan input (Bounds.finiteReal_spec input x value).1
68    simp_all
69  simp only [Results, nan, ni, pi, nv, true_and, false_and, false_or]
70  simp

72theorem neg_inf_iff (output : Word) : Results 0xff800000 output ↔ output = 0 := by
73  have hn : isNaN (0xff800000 : Word) = false := by decide
74  simp only [Results, hn, finiteReal_none _ (by decide : isFinite (0xff800000 : Word) = false)]
75  norm_num

77theorem pos_inf_iff (output : Word) : Results 0x7f800000 output ↔ output = 0x7f800000 := by
78  have hn : isNaN (0x7f800000 : Word) = false := by decide
79  simp only [Results, hn, finiteReal_none _ (by decide : isFinite (0x7f800000 : Word) = false)]
80  norm_num

82/-- Numerical existence holds exactly on the advertised domain. -/
83theorem results_exists_iff (input : Word) : (∃ output, Results input output) ↔ Supported input := by
84  constructor
85  · rintro ⟨output, nan | negInf | posInf | ⟨x,value,nonpos,_⟩⟩
86    · exact Or.inl nan.1
87    · exact Or.inr (Or.inl negInf.1)
88    · exact Or.inr (Or.inr (Or.inl posInf.1))
89    · exact Or.inr (Or.inr (Or.inr ⟨x,value,nonpos⟩))
90  · intro supported
91    rcases supported with nan | negInf | posInf | ⟨x,value,nonpos⟩
92    · exact ⟨input, (nan_iff input input nan).mpr nan⟩
93    · subst input; exact ⟨0, (neg_inf_iff 0).mpr rfl⟩
94    · subst input; exact ⟨0x7f800000, (pos_inf_iff _).mpr rfl⟩
95    · by_cases zero : x = 0
96      · subst x; exact ⟨0x3f800000, (zero_iff _ _ value).mpr rfl⟩
97      · obtain ⟨output,h⟩ := ConservativeULP.exp2_results_exists x nonpos
98        exact ⟨output, (negative_iff _ _ x value (lt_of_le_of_ne nonpos zero)).mpr h⟩
99
100theorem results_supported {input output : Word} (allowed : Results input output) :
101    Supported input := (results_exists_iff input).mp ⟨output,allowed⟩
102
103/-- Convenience form used by instruction existence proofs. -/
104theorem results_exists (input : Word) (supported : Supported input) :
105    ∃ output, Results input output := (results_exists_iff input).mpr supported
106
107/-- Exact one belongs to the broad finite envelope, enabling a bridge while
108retaining the stronger exact-zero contract in this provider. -/
109theorem one_allowed : ConservativeULP.Results 1 0x3f800000 := by
110  have center := RoundedReal.word_preserves_value 0x3f800000 1 Approximate.one_real
111    (RoundedReal.unit_range 1 (by norm_num) (by norm_num))
112  refine ⟨1,1,center,Approximate.one_real,?_⟩
113  simp only [sub_self, abs_zero]
114  exact mul_nonneg (by norm_num) (le_max_of_le_left (Approximate.ulp_nonnegative 1))
115
116/-- Every permitted finite result feeds the previously proved argument bridge. -/
117theorem finiteExp2 (input output : Word) (x : ℝ) (value : finiteReal input = some x)
118    (allowed : Results input output) : ExpArgument.FiniteExp2 input output := by
119  obtain ⟨nonpos,h⟩ := (finite_iff input output x value).mp allowed
120  refine ⟨x,value,nonpos,?_⟩
121  by_cases zero : x = 0
122  · subst x
123    simp only [ite_true] at h
124    subst output
125    simpa using one_allowed
126  · simpa [zero] using h
127
128/-- All-result error, with output finiteness derived rather than hypothesized. -/
129theorem finite_error (input output : Word) (x : ℝ) (value : finiteReal input = some x)
130    (allowed : Results input output) :
131    ∃ z : ℝ, finiteReal output = some z ∧ |z-(2:ℝ)^x| ≤ 5/(2:ℝ)^24 := by
132  obtain ⟨a,ha,nonpos,h⟩ := finiteExp2 input output x value allowed
133  have same : a = x := Option.some.inj (ha.symm.trans value)
134  subst a
135  exact ConservativeULP.exp2_error x output nonpos h
